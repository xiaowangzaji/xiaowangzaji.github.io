C++11 新特性

约 1304 字大约 4 分钟

C++C++11

2025-04-25

C++11 是 C++ 的一次重大更新，引入了许多新特性。

1. 自动类型推导

auto 关键字

auto 可以自动推导变量类型：

auto a = 10;           // int
auto b = 3.14;         // double
auto c = "hello";     // const char*
auto d = vector<int>(); // vector<int>

// 配合迭代器使用
vector<int> v = {1, 2, 3};
for (auto it = v.begin(); it != v.end(); it++) {
    cout << *it << endl;
}

// 范围 for 循环
for (auto& num : v) {
    num *= 2;
}

decltype 关键字

decltype 获取表达式的类型：

int x = 5;
decltype(x) y = 10;    // y 是 int

vector<int> v;
decltype(v)::iterator it = v.begin();  // 获取 vector<int>::iterator

constexpr（C++14）

编译时常量：

constexpr int square(int x) {
    return x * x;
}

int arr[square(5)];  // 编译时确定大小为 25

2. Lambda 表达式

基本语法

[capture](parameters) -> return_type {
    // 函数体
}

示例

// 无参数无返回值
auto f1 = []() { cout << "Hello" << endl; };
f1();

// 有参数
auto f2 = [](int a, int b) { return a + b; };
cout << f2(1, 2) << endl;  // 3

// 指定返回类型
auto f3 = [](int a, int b) -> double { return a / b; };

捕获列表

int x = 10;
int y = 20;

// 捕获值（复制）
auto f1 = [x]() { cout << x << endl; };

// 捕获引用
auto f2 = [&x]() { x = 100; };

// 捕获 this
auto f3 = [this]() { cout << member << endl; };

// 捕获所有（值）
auto f4 = [=]() { return x + y; };

// 捕获所有（引用）
auto f5 = [&]() { return x + y; };

// 混合捕获
auto f6 = [x, &y]() { y = x; };

使用场景

// STL 算法中使用
vector<int> v = {5, 2, 8, 1, 9};
sort(v.begin(), v.end(), [](int a, int b) { return a > b; });

// std::function
function<int(int, int)> add = [](int a, int b) { return a + b; };

3. 智能指针

独占所有权的 unique_ptr

#include <memory>

// 创建
unique_ptr<int> p1(new int(10));
unique_ptr<int> p2 = make_unique<int>(20);  // C++14 推荐

// 使用
*p1 = 100;
cout << *p1 << endl;

// 转移所有权
unique_ptr<int> p3 = move(p1);  // p1 变为空

// 释放
p3.reset();          // 释放内存
p3.release();       // 释放控制权，返回原始指针

// 数组
unique_ptr<int[]> arr(new int[10]);
arr[0] = 1;

共享所有权的 shared_ptr

shared_ptr<int> p1 = make_shared<int>(10);
shared_ptr<int> p2 = p1;  // 引用计数变为 2

cout << p1.use_count() << endl;  // 2

// 赋值
p2.reset();  // 引用计数减 1

// 访问
cout << *p1 << endl;
cout << p1.use_count() << endl;  // 1

弱引用的 weak_ptr

shared_ptr<int> p1 = make_shared<int>(10);
weak_ptr<int> w1 = p1;  // 不增加引用计数

if (auto p = w1.lock()) {
    cout << *p << endl;
} else {
    cout << "对象已销毁" << endl;
}

w1.reset();  // 释放弱引用

使用场景

// 工厂函数返回对象
unique_ptr<Shape> createCircle() {
    return make_unique<Circle>(1.0);
}

// 管理动态数组
unique_ptr<vector<int>> v = make_unique<vector<int>>(10);

// 避免循环引用
class Node {
public:
    shared_ptr<Node> next;
    weak_ptr<Node> prev;  // 使用 weak_ptr 避免循环
};

4. 移动语义

右值引用

int&& r = 10;  // 右值引用

void f(int&& x) {
    cout << x << endl;
}

f(20);  // 传入临时对象

移动构造函数

class Buffer {
private:
    int* data;
    size_t size;

public:
    // 移动构造函数
    Buffer(Buffer&& other) noexcept {
        data = other.data;
        size = other.size;
        other.data = nullptr;
        other.size = 0;
    }

    // 移动赋值运算符
    Buffer& operator=(Buffer&& other) noexcept {
        if (this != &other) {
            delete[] data;
            data = other.data;
            size = other.size;
            other.data = nullptr;
            other.size = 0;
        }
        return *this;
    }
};

std::move

vector<int> v1 = {1, 2, 3};
vector<int> v2 = move(v1);  // v1 变为空

// 在函数中返回局部对象
vector<int> createVector() {
    vector<int> v = {1, 2, 3};
    return move(v);  // 强制移动，避免拷贝
}

// emplace 使用 move
vector<string> v;
v.emplace_back("hello");  // 直接构造
v.push_back(move(s));     // 移动构造

5. 右值引用与完美转发

引用折叠规则

// T& &  -> T&
// T& &&  -> T&
// T&& &  -> T&
// T&& && -> T&&

完美转发

template<typename T>
void wrapper(T&& arg) {
    process(arg);  // 保持原始值类别
}

// 使用 forward
template<typename T>
void wrapper(T&& arg) {
    process(std::forward<T>(arg));
}

万能引用

template<typename T>
void func(T&& value) {}  // T&& 是万能引用

func(10);    // T 为 int，参数为 int&&
func(x);     // T 为 int&，参数为 int&
func(str);   // T 为 string，参数为 string&&

6. 标准多线程库

std::thread

#include <thread>
#include <mutex>
#include <atomic>

// 创建线程
void worker(int id) {
    cout << "Thread " << id << " running" << endl;
}

thread t1(worker, 1);
thread t2(worker, 2);
t1.join();
t2.join();

std::mutex

mutex m;
int counter = 0;

void increment() {
    m.lock();
    counter++;
    m.unlock();
}

// 或使用 lock_guard
void increment2() {
    lock_guard<mutex> lock(m);
    counter++;
}

std::atomic

atomic<int> counter(0);  // 原子操作

void increment() {
    counter.fetch_add(1);  // 原子加 1
}

stdfuture

#include <future>

int compute(int x) {
    return x * x;
}

// 异步执行
future<int> fut = async(compute, 10);
cout << fut.get() << endl;  // 等待并获取结果

条件变量

mutex mtx;
condition_variable cv;
bool ready = false;

void worker() {
    unique_lock<mutex> lock(mtx);
    cv.wait(lock, [] { return ready; });
    cout << "Work done" << endl;
}

void notify() {
    {
        lock_guard<mutex> lock(mtx);
        ready = true;
    }
    cv.notify_one();
}

7. nullptr 关键字

替代 NULL

void f(int);
void f(char*);

f(0);      // 模糊：调用 f(int)
f(NULL);   // 可能仍是 int
f(nullptr);  // 明确调用 f(char*)

类型安全

int* p = nullptr;  // 正确：空指针
void* v = nullptr;  // 也正确

// nullptr 有专门的类型
nullptr_t np = nullptr;

其他 C++11 特性

列表初始化

int arr[] = {1, 2, 3};
vector<int> v = {1, 2, 3};
map<string, int> m = {{"a", 1}, {"b", 2}};

范围 for 循环

vector<int> v = {1, 2, 3};
for (int x : v) {
    cout << x << endl;
}

委托构造函数

class A {
    int x, y;
public:
    A() : A(0, 0) {}
    A(int x) : A(x, 0) {}
    A(int x, int y) : x(x), y(y) {}
};

虚函数 override

class Base {
    virtual void f() {}
};

class Derived : Base {
    void f() override {}  // 明确标识重写
    // void f(int) override {}  // 错误：不是重写
};

final 关键字

class Base final {};  // 禁止继承

class Derived : Base {};  // 错误：不能继承

class A {
    virtual void f() final {}  // 禁止重写
};
