Python 编码规范

概述

遵循良好的编码规范可以让代码更易读、易维护。本章节介绍 Python 的编码规范和最佳实践，主要参考 PEP 8 风格指南。

一、代码布局

1.1 缩进

# ✅ 正确：使用 4 个空格
def hello():
    print("Hello")

# ❌ 错误：不要使用 Tab
def hello():
	print("Hello")
# 续行应该对齐
# 方式1：垂直对齐
foo = long_function_name(
    var_one, var_two,
    var_three, var_four)

# 方式2：4空格缩进
foo = long_function_name(
    var_one, var_two,
    var_three, var_four)

1.2 行长度

# 每行不超过 79 个字符（代码）
# 每行不超过 72 个字符（注释）

# 对于长字符串，可以使用括号续行
text = (
    "这是一个很长的字符串，"
    "可以分成多行书写"
)

1.3 空行

# 函数之间用两个空行
def func_one():
    pass


def func_two():
    pass


# 类之间用两个空行
class ClassOne:
    pass


class ClassTwo:
    pass


# 方法之间用一个空行
class MyClass:
    def method_one(self):
        pass

    def method_two(self):
        pass

1.4 导入

# ✅ 正确：按顺序导入
# 1. 标准库
import os
import sys

# 2. 第三方库
import requests
import numpy as np

# 3. 本地模块
from mypackage import mymodule

# 每个导入占一行
import os
import sys

# ❌ 错误：不要这样写
import os, sys

# 可以使用相对导入
from . import mymodule
from ..package import module

二、命名约定

2.1 变量命名

# ✅ 变量使用小写下划线（snake_case）
user_name = "张三"
is_active = True
total_count = 0

# ❌ 避免使用单字母（除非是循环变量）
# 不好
k = 0
# 好
for key in keys:
    pass

# 常量使用全大写
MAX_SIZE = 100
DEFAULT_TIMEOUT = 30

2.2 函数命名

# ✅ 函数名使用小写下划线
def get_user_info():
    pass

def calculate_total():
    pass

# 私有函数使用下划线前缀
def _private_function():
    pass

2.3 类命名

# ✅ 类名使用首字母大写（CapWords）
class UserInfo:
    pass

class MyClass:
    pass

# 异常类以 Error 结尾
class ValueError(Exception):
    pass

# 私有类使用下划线前缀
class _PrivateClass:
    pass

2.4 模块和包命名

# 模块名使用小写下划线
import my_module
import http_client

# 包名使用小写
import mypackage

三、注释规范

3.1 块注释

# 这是一个块注释
# 用于解释下面这段代码的作用
# 通常放在代码之前
def calculate(x, y):
    # 计算总和
    total = x + y
    return total

3.2 行内注释

# 谨慎使用行内注释
x = x + 1  # 补偿边界

3.3 文档字符串

def function_with_types(arg1, arg2):
    """在函数的文档字符串中说明参数类型和返回值。

    Args:
        arg1: 参数1的描述
        arg2: 参数2的描述

    Returns:
        返回值的描述

    Raises:
        ValueError: 何时抛出此异常
    """
    pass


class MyClass:
    """类的简单描述。

    类的详细描述（可选）。

    Attributes:
        attr1: 属性1的描述
        attr2: 属性2的描述
    """

    def method(self):
        """方法的简单描述。"""
        pass

3.4 文档字符串风格

# Google 风格
def func(arg1, arg2):
    """Summary line.

    Args:
        arg1 (int): Description of arg1
        arg2 (str): Description of arg2

    Returns:
        bool: Description of return value
    """
    pass


# NumPy 风格
def func(arg1, arg2):
    """
    Summary line.

    Parameters
    ----------
    arg1 : int
        Description of arg1
    arg2 : str
        Description of arg2

    Returns
    -------
    bool
        Description of return value
    """
    pass

四、字符串引号

# ✅ 单引号和双引号都可以
name = '张三'
name = "张三"

# 文档字符串使用双引号
"""这是文档字符串"""


# 保持一致性
# 如果字符串中有双引号，使用单引号
message = "It's a good day"

# 如果字符串中有单引号，使用双引号
message = 'I said "hello"'

五、空格使用

5.1 运算符周围

# ✅ 正确
i = i + 1
submitted += 1
x = -1
y = 2 * 3
a = b * c + d

# ❌ 错误
i=i+1
submitted+=1

5.2 函数参数

# ✅ 正确
def complex(real, imag):
    pass

# ❌ 错误
def complex(real,imag):
    pass

5.3 赋值

# ✅ 正确
x = 1
y = 2
long_variable = 3

# ❌ 错误
x=1
y=2
long_variable =3

六、代码设计

6.1 函数设计

# ✅ 函数应该简短，只做一件事
def send_email(to, subject, body):
    """发送邮件"""
    # 发送邮件逻辑
    pass

# ❌ 避免函数过长或做多件事
def process_user():  # 不好：做太多事
    # 验证用户
    # 获取用户数据
    # 发送邮件
    # 更新数据库
    # ...
    pass


# ✅ 使用默认参数
def get_user(name, active=True):
    pass


# ✅ 使用 None 作为可变默认参数
def append_to(element, to=None):
    if to is None:
        to = []
    to.append(element)
    return to


# ❌ 错误：不要使用可变对象作为默认参数
def append_to(element, to=[]):  # 危险！
    to.append(element)
    return to

6.2 类设计

class MyClass:
    """类的文档字符串"""

    # 公共属性
    public_attr = "公共属性"

    # 私有属性
    _private_attr = "私有属性"

    def __init__(self):
        """初始化方法"""
        pass

    def public_method(self):
        """公共方法"""
        pass

    def _private_method(self):
        """私有方法"""
        pass


# ✅ 使用 __slots__ 节省内存
class Point:
    __slots__ = ['x', 'y']

    def __init__(self, x, y):
        self.x = x
        self.y = y

6.3 异常处理

# ✅ 具体异常捕获
try:
    result = divide(10, 2)
except ZeroDivisionError:
    print("不能除以零")

# ✅ 使用 else 子句
try:
    result = int(input())
except ValueError:
    print("无效输入")
else:
    print(f"你输入了: {result}")

# ✅ 异常要记录或重新抛出
try:
    dangerous_operation()
except SomeError as e:
    logging.error(f"操作失败: {e}")
    raise  # 重新抛出或自定义处理

7.3 类型别名

from typing import List, Tuple

# 简单类型别名
Vector = List[float]
ConnectionDict = Dict[str, str]

UserDict = Dict[str, Union[str, int, bool]]
Matrix = List[List[int]]

九、代码格式化工具

9.1 Black（推荐）

# 安装
pip install black

# 使用
black mymodule.py

# 配置 pyproject.toml
[tool.black]
line-length = 88
target-version = ['py38']
include = '\.pyi?$'

9.2 Ruff（推荐）

# 安装
pip install ruff

# 检查
ruff check mymodule.py

# 自动修复
ruff check --fix mymodule.py

# 格式化
ruff format mymodule.py
