Python 并发编程

概述

并发编程允许程序同时执行多个任务，提升程序效率。Python 提供多种并发方式：多线程（I/O 密集型）、多进程（CPU 密集型）和异步编程。

一、多线程 threading

1.1 基础使用

import threading
import time

def worker(name, duration):
    """工作函数"""
    print(f"线程 {name} 开始")
    time.sleep(duration)
    print(f"线程 {name} 结束")

# 创建线程
t1 = threading.Thread(target=worker, args=('A', 2))
t2 = threading.Thread(target=worker, args=('B', 1))

# 启动线程
t1.start()
t2.start()

# 等待线程结束
t1.join()
t2.join()

print("所有线程完成")

代码说明：创建两个线程同时执行任务，通过 join() 等待完成。

1.2 继承 Thread 类

import threading

class MyThread(threading.Thread):
    def __init__(self, name, duration):
        super().__init__()
        self.name = name
        self.duration = duration
    
    def run(self):
        """线程执行的方法"""
        print(f"线程 {self.name} 开始")
        time.sleep(self.duration)
        print(f"线程 {self.name} 结束")

# 使用
threads = []
for i in range(3):
    t = MyThread(f"T{i}", 2-i)
    threads.append(t)
    t.start()

for t in threads:
    t.join()

1.3 线程同步 - Lock

import threading

# 共享资源
counter = 0
lock = threading.Lock()

def worker(lock):
    global counter
    for _ in range(100000):
        with lock:  # 获取锁
            counter += 1

# 创建多个线程
threads = [threading.Thread(target=worker, args=(lock,)) for _ in range(10)]

for t in threads:
    t.start()
for t in threads:
    t.join()

print(f"计数器: {counter}")  # 应该是 1000000

1.4 线程同步 - RLock（可重入锁）

import threading

lock = threading.RLock()

def worker():
    with lock:
        print("获取锁")
        with lock:  # 可重入，同一线程可多次获取
            print("再次获取锁")
        print("释放内层锁")
    print("释放外层锁")

worker()

1.5 线程同步 - Condition

import threading

class Buffer:
    def __init__(self):
        self.buffer = []
        self.condition = threading.Condition()
        self.max_size = 5
    
    def put(self, item):
        with self.condition:
            while len(self.buffer) >= self.max_size:
                self.condition.wait()  # 等待消费者
            self.buffer.append(item)
            print(f"放入: {item}")
            self.condition.notify()  # 通知消费者
    
    def get(self):
        with self.condition:
            while len(self.buffer) == 0:
                self.condition.wait()  # 等待生产者
            item = self.buffer.pop(0)
            print(f"取出: {item}")
            self.condition.notify()  # 通知生产者
            return item

1.6 Semaphore（信号量）

import threading
import time

# 限制同时访问的线程数
semaphore = threading.Semaphore(3)

def worker(name):
    with semaphore:
        print(f"线程 {name} 获取许可")
        time.sleep(1)
        print(f"线程 {name} 释放许可")

# 5个线程，但只有3个能同时执行
threads = [threading.Thread(target=worker, args=(i,)) for i in range(5)]
for t in threads:
    t.start()
for t in threads:
    t.join()

1.7 ThreadLocal

import threading

# 线程局部变量
local_data = threading.local()

def worker():
    # 每个线程独立的变量
    local_data.value = threading.current_thread().name
    print(f"线程 {threading.current_thread().name} 的值: {local_data.value}")

threads = [threading.Thread(target=worker) for _ in range(3)]
for t in threads:
    t.start()
for t in threads:
    t.join()

二、多进程 multiprocessing

2.1 基础使用

import multiprocessing
import time

def worker(name, duration):
    """工作函数"""
    print(f"进程 {name} 开始")
    time.sleep(duration)
    print(f"进程 {name} 结束")

# 创建进程
if __name__ == '__main__':
    p1 = multiprocessing.Process(target=worker, args=('A', 2))
    p2 = multiprocessing.Process(target=worker, args=('B', 1))

    p1.start()
    p2.start()

    p1.join()
    p2.join()

    print("所有进程完成")

注意

Windows 下 multiprocessing 需要在 if __name__ == '__main__': 保护下运行。

2.2 进程池

import multiprocessing
import time

def worker(x):
    """工作函数"""
    time.sleep(1)
    return x * x

if __name__ == '__main__':
    # 创建进程池
    with multiprocessing.Pool(processes=4) as pool:
        # map - 同步返回结果
        result = pool.map(worker, [1, 2, 3, 4])
        print(result)  # [1, 4, 9, 16]
        
        # apply_async - 异步提交任务
        result = pool.apply_async(worker, (5,))
        print(result.get())  # 25
        
        # map_async - 异步 map
        async_result = pool.map_async(worker, [6, 7, 8])
        print(async_result.get())  # [36, 49, 64]

2.3 进程间通信

import multiprocessing

def producer(queue):
    """生产者"""
    for i in range(5):
        queue.put(i)
    queue.put(None)  # 发送结束信号

def consumer(queue):
    """消费者"""
    while True:
        item = queue.get()
        if item is None:
            break
        print(f"消费: {item}")

if __name__ == '__main__':
    queue = multiprocessing.Queue()
    
    p1 = multiprocessing.Process(target=producer, args=(queue,))
    p2 = multiprocessing.Process(target=consumer, args=(queue,))
    
    p1.start()
    p2.start()
    
    p1.join()
    p2.join()

2.4 进程间共享内存

import multiprocessing

def worker(shared_value, shared_array, lock):
    """使用共享内存"""
    with lock:
        shared_value.value += 1
        shared_array[0] += 10

if __name__ == '__main__':
    # 共享值
    shared_value = multiprocessing.Value('i', 0)  # 'i' 表示整数
    lock = multiprocessing.Lock()
    
    # 共享数组
    shared_array = multiprocessing.Array('i', [0, 0, 0])
    
    # 创建进程
    processes = [
        multiprocessing.Process(target=worker, args=(shared_value, shared_array, lock))
        for _ in range(5)
    ]
    
    for p in processes:
        p.start()
    for p in processes:
        p.join()
    
    print(f"共享值: {shared_value.value}")  # 5
    print(f"共享数组: {list(shared_array)}")  # [50, 0, 0]

三、异步编程 asyncio

3.1 基础使用

import asyncio

async def task(name, duration):
    """异步任务"""
    print(f"{name} 开始")
    await asyncio.sleep(duration)
    print(f"{name} 结束")
    return f"{name} 结果"

async def main():
    # 创建任务
    t1 = asyncio.create_task(task('A', 2))
    t2 = asyncio.create_task(task('B', 1))
    
    # 等待所有任务
    results = await asyncio.gather(t1, t2)
    print(f"结果: {results}")

# 运行
asyncio.run(main())

代码说明：使用 async/await 语法定义异步任务，通过 asyncio.run() 运行。

3.2 async/await

import asyncio

async def fetch_data(url):
    """模拟异步获取数据"""
    print(f"开始获取: {url}")
    await asyncio.sleep(1)  # 模拟网络请求
    return {"data": url}

async def main():
    # 顺序执行
    data1 = await fetch_data("api1")
    data2 = await fetch_data("api2")
    print(data1, data2)
    
    # 并发执行
    results = await asyncio.gather(
        fetch_data("api1"),
        fetch_data("api2"),
        fetch_data("api3")
    )
    print(results)

asyncio.run(main())

3.3 asyncio 锁

import asyncio

async def worker(name, lock):
    """使用异步锁"""
    async with lock:
        print(f"{name} 获取锁")
        await asyncio.sleep(1)
        print(f"{name} 释放锁")

async def main():
    lock = asyncio.Lock()
    
    await asyncio.gather(
        worker("A", lock),
        worker("B", lock),
        worker("C", lock)
    )

asyncio.run(main())

3.4 异步生成器

import asyncio

async def async_generator(n):
    """异步生成器"""
    for i in range(n):
        await asyncio.sleep(0.5)
        yield i

async def main():
    async for value in async_generator(5):
        print(value)

asyncio.run(main())

3.5 异步队列

import asyncio

async def producer(queue, n):
    for i in range(n):
        await queue.put(i)
        print(f"生产: {i}")
    await queue.put(None)  # 结束信号

async def consumer(queue):
    while True:
        item = await queue.get()
        if item is None:
            break
        print(f"消费: {item}")

async def main():
    queue = asyncio.Queue()
    
    await asyncio.gather(
        producer(queue, 5),
        consumer(queue)
    )

asyncio.run(main())

四、对比与选择

4.1 适用场景

方式适用场景原因
多线程I/O 密集型GIL 允许线程等待 I/O 时释放锁
多进程CPU 密集型绕过 GIL，每个进程独立
异步高并发 I/O单线程内高效轮询事件

4.2 GIL 问题

import threading
import time

# CPU 密集型任务 - 多线程无法利用多核
def cpu_task():
    result = 0
    for i in range(10000000):
        result += i
    return result

# 使用多线程
threads = [threading.Thread(target=cpu_task) for _ in range(4)]
start = time.time()
for t in threads:
    t.start()
for t in threads:
    t.join()
print(f"多线程: {time.time() - start:.2f}秒")

# 使用多进程
import multiprocessing
processes = [multiprocessing.Process(target=cpu_task) for _ in range(4)]
start = time.time()
for p in processes:
    p.start()
for p in processes:
    p.join()
print(f"多进程: {time.time() - start:.2f}秒")

4.3 选择建议

# I/O 密集型 - 优先使用 asyncio 或线程
# 例子: 网络请求、文件读写、数据库查询

# CPU 密集型 - 使用多进程
# 例子: 数学计算、图像处理、机器学习

# 混合型 - 分情况处理
# I/O 部分用 asyncio，CPU 部分用进程池

五、实际案例

5.1 异步 HTTP 请求

import asyncio
import aiohttp

async def fetch_url(session, url):
    async with session.get(url) as response:
        return await response.text()

async def main():
    urls = [
        'https://httpbin.org/delay/1',
        'https://httpbin.org/delay/2',
        'https://httpbin.org/delay/1'
    ]
    
    async with aiohttp.ClientSession() as session:
        tasks = [fetch_url(session, url) for url in urls]
        results = await asyncio.gather(*tasks)
        print(f"完成 {len(results)} 个请求")

asyncio.run(main())

5.2 并行处理数据

import multiprocessing
from concurrent.futures import ProcessPoolExecutor

def process_item(item):
    """处理单个数据项"""
    # 耗时计算
    return item * 2

def parallel_process(items, workers=4):
    """并行处理数据"""
    with ProcessPoolExecutor(max_workers=workers) as executor:
        results = list(executor.map(process_item, items))
    return results

if __name__ == '__main__':
    data = list(range(100))
    results = parallel_process(data)
    print(f"处理了 {len(results)} 项")

5.3 带超时的并发

import asyncio

async def long_task():
    await asyncio.sleep(5)
    return "完成"

async def main():
    try:
        # 等待任务，最多3秒超时
        result = await asyncio.wait_for(long_task(), timeout=3)
        print(result)
    except asyncio.TimeoutError:
        print("任务超时")

asyncio.run(main())
