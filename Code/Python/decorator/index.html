Python 装饰器

概述

装饰器是 Python 的强大特性，允许在不修改原函数的情况下增强其功能。本质上是一个返回函数的函数。

一、闭包

1.1 什么是闭包

闭包是指内部函数引用外部作用域变量的现象。

def outer():
    x = 10  # 外部变量
    
    def inner():
        print(x)  # 引用外部变量
        return x
    
    return inner

# 测试
func = outer()
print(func())  # 10

代码说明：inner 函数引用了 outer 的局部变量 x，形成闭包。

1.2 闭包案例

def make_multiplier(factor):
    """创建乘法函数"""
    def multiply(number):
        return number * factor
    return multiply

times3 = make_multiplier(3)
times5 = make_multiplier(5)

print(times3(10))  # 30
print(times5(10))  # 50

二、基础装饰器

2.1 无参装饰器

def my_decorator(func):
    """装饰器函数"""
    def wrapper():
        print("调用前")
        func()
        print("调用后")
    return wrapper

@my_decorator
def say_hello():
    print("Hello!")

# 调用
say_hello()

输出：

调用前
Hello!
调用后

2.2 装饰器原理

# 以下两种写法等价

# 方式1：使用 @ 语法
@my_decorator
def say_hello():
    print("Hello!")

# 方式2：手动调用
def say_hello():
    print("Hello!")
say_hello = my_decorator(say_hello)

2.3 带参数的原函数

def my_decorator(func):
    def wrapper(*args, **kwargs):
        print("调用前")
        result = func(*args, **kwargs)
        print("调用后")
        return result
    return wrapper

@my_decorator
def greet(name):
    return f"Hello, {name}!"

print(greet("张三"))  # 自动传递参数

三、带参数装饰器

3.1 双层装饰器

def repeat(times):
    """重复执行指定次数"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(times=3)
def say_hello():
    print("Hello!")

say_hello()  # Hello! 打印 3 次

3.2 日志装饰器

import logging
from functools import wraps

def log(func):
    """记录函数调用日志"""
    @wraps(func)  # 保留原函数信息
    def wrapper(*args, **kwargs):
        logging.info(f"调用 {func.__name__}")
        result = func(*args, **kwargs)
        logging.info(f"{func.__name__} 执行完成")
        return result
    return wrapper

@log
def add(a, b):
    """加法函数"""
    return a + b

print(add(3, 5))
print(add.__name__)  # add（保留原函数名）
print(add.__doc__)  # 加法函数（保留原文档）

提示

使用 @wraps 保留原函数的 __name__、__doc__ 等属性。

3.3 计时装饰器

import time
from functools import wraps

def timer(func):
    """函数执行时间"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 执行耗时: {end - start:.4f}秒")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
    return "完成"

slow_function()  # slow_function 执行耗时: 1.0012秒

四、类装饰器

4.1 函数作为类装饰器

def singleton(cls):
    """单例装饰器"""
    instances = {}
    
    @wraps(cls)
    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    
    return get_instance

@singleton
class Database:
    def __init__(self):
        print("创建数据库连接")
    
    def query(self, sql):
        return f"执行: {sql}"

# 测试
db1 = Database()
db2 = Database()
print(db1 is db2)  # True

4.2 类作为装饰器

class CountCalls:
    """统计函数调用次数"""
    def __init__(self, func):
        self.func = func
        self.count = 0
        wraps(func)(self)
    
    def __call__(self, *args, **kwargs):
        self.count += 1
        print(f"调用次数: {self.count}")
        return self.func(*args, **kwargs)

@CountCalls
def say_hello():
    return "Hello"

say_hello()  # 调用次数: 1
say_hello()  # 调用次数: 2

五、内置装饰器

5.1 @property

class Circle:
    def __init__(self, radius):
        self.radius = radius
    
    @property
    def diameter(self):
        return self.radius * 2
    
    @property
    def area(self):
        return 3.14159 * self.radius ** 2

c = Circle(5)
print(c.diameter)  # 10（像属性一样访问）
print(c.area)     # 78.53975

5.2 @classmethod

class Math:
    @classmethod
    def add(cls, a, b):
        return a + b

# 直接通过类调用
result = Math.add(3, 5)
print(result)  # 8

5.3 @staticmethod

class StringUtils:
    @staticmethod
    def is_palindrome(s):
        return s == s[::-1]

print(StringUtils.is_palindrome("abcba"))  # True

6.2 装饰器传参

def retry(max_attempts=3, delay=1):
    """重试装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    time.sleep(delay)
            return wrapper
        return decorator
    return decorator

@retry(max_attempts=3, delay=1)
def fetch_data():
    import random
    if random.random() > 0.8:
        return "数据"
    raise ConnectionError("网络错误")

fetch_data()

6.3 装饰器类

class Timer:
    def __init__(self, func):
        self.func = func
        self.time = 0
    
    def __call__(self, *args, **kwargs):
        import time
        start = time.time()
        result = self.func(*args, **kwargs)
        self.time = time.time() - start
        return result

@Timer
def calculate():
    sum(range(1000000))
    return "完成"

calculate()
print(f"耗时: {calculate.time:.4f}秒")

七、实际应用

7.1 缓存装饰器

from functools import wraps

def memoize(func):
    """缓存结果"""
    cache = {}
    
    @wraps(func)
    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
        return cache[args]
    
    wrapper.cache = cache
    return wrapper

@memoize
def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

print(fibonacci(100))  # 快速计算

7.3 性能限制

from functools import wraps
import time

def rate_limit(calls=5, period=60):
    """限流装饰器"""
    def decorator(func):
        calls_history = []
        
        @wraps(func)
        def wrapper(*args, **kwargs):
            now = time.time()
            # 清理过期记录
            calls_history[:] = [t for t in calls_history if now - t < period]
            
            if len(calls_history) >= calls:
                raise RuntimeError(f"调用过于频繁，请在 {period} 秒后重试")
            
            calls_history.append(now)
            return func(*args, **kwargs)
        
        return wrapper
    return decorator

@rate_limit(calls=3, period=60)
def send_message(msg):
    return f"发送: {msg}"
