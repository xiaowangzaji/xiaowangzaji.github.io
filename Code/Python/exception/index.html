Python 异常处理

概述

异常处理是保证程序健壮性的关键。Python 提供了完善的异常处理机制，让我们能够优雅地处理运行时的错误，而不是让程序直接崩溃。

一、基础异常处理

1.1 try-except 基本语法

try:
    # 可能发生异常的代码
    result = 10 / 0
except ZeroDivisionError:
    # 处理特定异常
    print("不能除以零！")

代码说明：尝试执行可能出错的代码，如果发生 ZeroDivisionError 异常，执行 except 块中的代码。

1.2 捕获异常信息

try:
    result = 10 / 0
except ZeroDivisionError as e:
    # 获取异常信息
    print(f"错误类型: {type(e).__name__}")
    print(f"错误信息: {e}")

1.3 多个 except

try:
    # 可能多种异常
    num = int(input("请输入数字: "))
    result = 10 / num
except ValueError:
    print("输入的不是有效数字！")
except ZeroDivisionError:
    print("不能除以零！")

1.4 捕获所有异常

try:
    result = 10 / 0
except Exception as e:
    print(f"发生错误: {e}")

注意

尽量避免使用裸 except:，会捕获所有异常包括 KeyboardInterrupt。

1.5 try-except-else

try:
    num = int(input("请输入数字: "))
    result = 10 / num
except ValueError:
    print("输入无效")
except ZeroDivisionError:
    print("不能除以零")
else:
    # 只有没有异常时才执行
    print(f"结果是: {result}")

1.6 try-finally

try:
    file = open('data.txt', 'r')
    content = file.read()
except FileNotFoundError:
    print("文件不存在")
finally:
    # 无论是否有异常都会执行
    print("清理资源...")
    # file.close()  # 确保关闭文件

二、抛出异常

2.1 raise 语句

def divide(a, b):
    if b == 0:
        raise ValueError("除数不能为零")
    return a / b

try:
    result = divide(10, 0)
except ValueError as e:
    print(f"捕获到异常: {e}")

2.2 重新抛出异常

def outer():
    try:
        inner()
    except ValueError as e:
        print(f"Outer 捕获: {e}")
        # 重新抛出
        raise

def inner():
    raise ValueError("内部错误")

try:
    outer()
except ValueError as e:
    print(f"最终捕获: {e}")

2.3 异常链

try:
    # Python 3.10+ 使用 raise from
    raise ValueError("原始错误") from TypeError("原因错误")
except ValueError as e:
    print(f"主异常: {e}")
    print(f"原因异常: {e.__cause__}")

三、自定义异常

3.1 创建异常类

class ValidationError(Exception):
    """验证错误异常"""
    def __init__(self, field, message):
        self.field = field
        self.message = message
        super().__init__(f"{field}: {message}")

def validate_age(age):
    if age < 0:
        raise ValidationError("age", "年龄不能为负数")
    if age > 150:
        raise ValidationError("age", "年龄不合法")
    return True

try:
    validate_age(-5)
except ValidationError as e:
    print(f"验证失败: {e.field} - {e.message}")

3.2 异常层次结构

# 自定义异常基类
class AppError(Exception):
    """应用基础异常"""
    pass

# 具体异常
class DataError(AppError):
    """数据错误"""
    pass

class NetworkError(AppError):
    """网络错误"""
    pass

class DatabaseError(NetworkError):
    """数据库错误"""
    pass

四、断言

4.1 assert 语句

# 基本语法
assert 条件, 错误信息

# 示例
def calculate_average(numbers):
    assert len(numbers) > 0, "列表不能为空"
    return sum(numbers) / len(numbers)

print(calculate_average([1, 2, 3, 4, 5]))  # 3.0
print(calculate_average([]))  # AssertionError: 列表不能为空

def withdraw(balance, amount):
    """取款函数"""
    assert amount > 0, "取款金额必须为正"
    assert amount <= balance, "余额不足"
    return balance - amount

balance = 1000
# 测试
print(withdraw(balance, 100))   # 900
# print(withdraw(balance, 1500)) # AssertionError: 余额不足

五、上下文管理器

5.1 类实现上下文管理器

class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None
    
    def __enter__(self):
        """进入上下文"""
        self.file = open(self.filename, self.mode)
        return self.file
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出上下文"""
        if self.file:
            self.file.close()
        # 返回 True 表示抑制异常
        return False

# 使用
with FileManager('test.txt', 'w') as f:
    f.write("Hello, World!")

5.2 contextlib 模块

from contextlib import contextmanager

@contextmanager
def managed_resource():
    """使用生成器创建上下文管理器"""
    print("获取资源")
    yield "资源对象"
    print("释放资源")

with managed_resource() as resource:
    print(f"使用: {resource}")

六、异常处理最佳实践

6.1 EAFP vs LBYL

# LBYL (Look Before You Leap) - 事前检查
if os.path.exists('file.txt'):
    with open('file.txt') as f:
        data = f.read()

# EAFP (Easier to Ask Forgiveness than Permission) - 异常处理
try:
    with open('file.txt') as f:
        data = f.read()
except FileNotFoundError:
    data = None

提示

Python 中推荐使用 EAFP 风格，代码更简洁。

6.2 异常捕获原则

# ✅ 好的做法：具体明确
try:
    result = divide(10, 2)
except ZeroDivisionError:
    print("除数不能为零")

# ❌ 不好的做法：过于宽泛
try:
    result = divide(10, 2)
except:
    print("出错了")

6.3 记录异常日志

import logging

logging.basicConfig(level=logging.ERROR)

def divide(a, b):
    try:
        return a / b
    except ZeroDivisionError as e:
        logging.error(f"除法错误: {a}/{b}, {e}")
        raise

divide(10, 0)

八、综合示例

8.1 数据验证

class ValidationException(Exception):
    """验证异常"""
    pass

def validate_user(username, email, age):
    """用户信息验证"""
    errors = []
    
    # 验证用户名
    if not username or len(username) < 3:
        errors.append("用户名至少3个字符")
    
    # 验证邮箱
    if '@' not in email:
        errors.append("邮箱格式无效")
    
    # 验证年龄
    if not isinstance(age, int) or age < 0 or age > 150:
        errors.append("年龄无效")
    
    if errors:
        raise ValidationException("; ".join(errors))
    
    return True

# 使用
try:
    validate_user("ab", "invalid", 25)
except ValidationException as e:
    print(f"验证失败: {e}")

8.2 重试机制

import time

def retry(max_attempts=3, delay=1):
    """重试装饰器"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"第 {attempt + 1} 次失败，重试中...")
                    time.sleep(delay)
            return wrapper
        return decorator
    return decorator

@retry(max_attempts=3)
def fetch_data():
    """模拟获取数据"""
    import random
    if random.random() > 0.7:
        return "数据"
    raise ConnectionError("网络错误")

# 测试
for _ in range(5):
    try:
        print(fetch_data())
    except ConnectionError as e:
        print(f"最终失败: {e}")
