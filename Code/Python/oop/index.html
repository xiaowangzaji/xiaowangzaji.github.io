Python 面向对象

概述

面向对象编程（OOP）是一种将数据和操作数据的方法封装在一起的编程范式。Python 作为一门面向对象的语言，完全支持类和对象的所有特性。

一、类与对象基础

1.1 定义类

class Person:
    """人类"""
    
    # 类属性
    species = "智人"
    
    # 初始化方法（构造方法）
    def __init__(self, name, age):
        # 实例属性
        self.name = name
        self.age = age
    
    # 实例方法
    def introduce(self):
        return f"我叫{self.name}，今年{self.age}岁"

代码说明：定义一个 Person 类，包含类属性、初始化方法和实例方法。

1.2 创建对象

# 创建对象
person = Person("张三", 25)

# 访问属性
print(person.name)  # 张三
print(person.age)   # 25

# 调用方法
print(person.introduce())  # 我叫张三，今年25岁

1.3 self 详解

class Dog:
    def __init__(self, name, breed):
        self.name = name      # self 指向当前对象
        self.breed = breed
    
    def bark(self):
        # self.name 访问当前对象的属性
        return f"{self.name} 正在汪汪叫"

dog = Dog("旺财", "金毛")
print(dog.bark())  # 旺财 正在汪汪叫

提示

self 代表当前对象的引用，在方法内部通过 self 访问对象的属性和方法。

二、属性与特性

2.1 类属性与实例属性

class Circle:
    # 类属性（所有实例共享）
    pi = 3.14159
    
    def __init__(self, radius):
        # 实例属性（每个对象独立）
        self.radius = radius
    
    def area(self):
        return Circle.pi * self.radius ** 2

# 测试
c1 = Circle(5)
c2 = Circle(10)

print(c1.area())  # 78.53975
print(c2.area())  # 314.159

2.2 私有属性

class BankAccount:
    def __init__(self, balance):
        # 私有属性：以双下划线开头
        self.__balance = balance
    
    def get_balance(self):
        """公开的查询方法"""
        return self.__balance
    
    def deposit(self, amount):
        """存款"""
        if amount > 0:
            self.__balance += amount
            return True
        return False
    
    def withdraw(self, amount):
        """取款"""
        if amount > 0 and amount <= self.__balance:
            self.__balance -= amount
            return True
        return False

account = BankAccount(1000)
print(account.get_balance())  # 1000
account.deposit(500)
print(account.get_balance())  # 1500
# account.__balance  # 报错：无法直接访问

注意

Python 的私有属性是"伪私有"，实际存储为 _类名__属性名。

2.3 @property 装饰器

class Person:
    def __init__(self, name, age):
        self.name = name
        self._age = age
    
    @property
    def age(self):
        """获取年龄"""
        return self._age
    
    @age.setter
    def age(self, value):
        """设置年龄，带验证"""
        if value < 0:
            raise ValueError("年龄不能为负数")
        self._age = value
    
    @age.deleter
    def age(self):
        """删除年龄"""
        del self._age

# 使用
p = Person("张三", 25)
print(p.age)      # 25（像访问属性一样）
p.age = 30       # 自动验证
# del p.age      # 删除属性

三、继承

3.1 基本继承

# 父类（基类）
class Animal:
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        return "发出声音"
    
    def eat(self, food):
        return f"吃{food}"

# 子类（派生类）
class Dog(Animal):
    def speak(self):
        return "汪汪汪"

class Cat(Animal):
    def speak(self):
        return "喵喵喵"

# 测试
dog = Dog("旺财")
cat = Cat("咪咪")

print(dog.speak())  # 汪汪汪
print(cat.speak())  # 喵喵喵
print(dog.eat("骨头"))  # 吃骨头

3.2 super() 调用父类

class Animal:
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def show_info(self):
        return f"{self.name}, {self.age}岁"

class Dog(Animal):
    def __init__(self, name, age, breed):
        # 调用父类的 __init__
        super().__init__(name, age)
        self.breed = breed
    
    def show_info(self):
        # 调用父类的方法
        base_info = super().show_info()
        return f"{base_info}, 品种: {self.breed}"

dog = Dog("旺财", 3, "金毛")
print(dog.show_info())  # 旺财, 3岁, 品种: 金毛

3.3 多继承

class Flyable:
    def fly(self):
        return "正在飞"

class Swimmable:
    def swim(self):
        return "正在游泳"

class Duck(Flyable, Swimmable):
    def quack(self):
        return "嘎嘎叫"

duck = Duck()
print(duck.fly())    # 正在飞
print(duck.swim())   # 正在游泳
print(duck.quack())  # 嘎嘎叫

提示

多继承时，方法解析顺序（MRO）从左到右，优先使用左边的父类方法。

3.4 方法重写

class Shape:
    def area(self):
        return 0

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    
    def area(self):
        return 3.14159 * self.radius ** 2

shapes = [Rectangle(3, 4), Circle(5), Rectangle(2, 3)]
for shape in shapes:
    print(f"面积: {shape.area()}")

四、多态

4.1 概念

多态是指同一种行为对于不同对象有不同表现形式。在 Python 中天然支持多态。

class Animal:
    def make_sound(self):
        pass

class Dog(Animal):
    def make_sound(self):
        return "汪汪汪"

class Cat(Animal):
    def make_sound(self):
        return "喵喵喵"

class Duck(Animal):
    def make_sound(self):
        return "嘎嘎嘎"

# 统一接口
def play_sound(animal):
    print(animal.make_sound())

# 多态体现
animals = [Dog(), Cat(), Duck()]
for animal in animals:
    play_sound(animal)

4.2 抽象基类

from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self):
        """计算面积"""
        pass
    
    @abstractmethod
    def perimeter(self):
        """计算周长"""
        pass

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)

# rect = Shape()  # 报错：抽象类不能直接实例化
rect = Rectangle(3, 4)
print(rect.area())       # 12
print(rect.perimeter())  # 14

五、类方法与静态方法

5.1 类方法

class Person:
    count = 0  # 类属性
    
    def __init__(self, name):
        self.name = name
        Person.count += 1
    
    @classmethod
    def get_count(cls):
        """类方法：访问类属性"""
        return f"已创建 {cls.count} 个人"

# 测试
p1 = Person("张三")
p2 = Person("李四")
print(Person.get_count())  # 已创建 2 个人

5.2 静态方法

class MathUtils:
    @staticmethod
    def add(a, b):
        """静态方法：不需要访问类或实例"""
        return a + b
    
    @staticmethod
    def is_even(n):
        return n % 2 == 0

# 测试
print(MathUtils.add(3, 5))      # 8
print(MathUtils.is_even(10))   # True

5.3 三种方法对比

class Example:
    class_attr = "类属性"
    
    def __init__(self):
        self.instance_attr = "实例属性"
    
    def instance_method(self):
        """实例方法：需要 self"""
        return "实例方法"
    
    @classmethod
    def class_method(cls):
        """类方法：需要 cls"""
        return f"类方法 - {cls.class_attr}"
    
    @staticmethod
    def static_method():
        """静态方法：不需要 self 或 cls"""
        return "静态方法"

六、特殊方法（魔术方法）

6.1 构造与销毁

class Database:
    def __new__(cls, *args, **kwargs):
        """创建对象前调用"""
        print("创建数据库连接...")
        return super().__new__(cls)
    
    def __init__(self, name):
        print(f"初始化数据库: {name}")
        self.name = name
    
    def __del__(self):
        """对象销毁时调用"""
        print(f"关闭数据库: {self.name}")

db = Database("myapp")  # 创建并初始化
del db                   # 销毁

6.2 字符串表示

class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def __str__(self):
        """str() 调用"""
        return f"Person({self.name}, {self.age})"
    
    def __repr__(self):
        """repr() 调用"""
        return f"Person(name='{self.name}', age={self.age})"

p = Person("张三", 25)
print(str(p))    # Person(张三, 25)
print(repr(p))   # Person(name='张三', age=25)

6.3 运算相关

class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, other):
        """加法 +"""
        return Vector(self.x + other.x, self.y + other.y)
    
    def __mul__(self, scalar):
        """乘法 *"""
        return Vector(self.x * scalar, self.y * scalar)
    
    def __repr__(self):
        return f"Vector({self.x}, {self.y})"

v1 = Vector(1, 2)
v2 = Vector(3, 4)
print(v1 + v2)    # Vector(4, 6)
print(v1 * 3)    # Vector(3, 6)

6.4 容器行为

class Stack:
    def __init__(self):
        self._items = []
    
    def push(self, item):
        self._items.append(item)
    
    def pop(self):
        return self._items.pop()
    
    def __len__(self):
        return len(self._items)
    
    def __getitem__(self, index):
        return self._items[index]
    
    def __repr__(self):
        return f"Stack({self._items})"

stack = Stack()
stack.push(1)
stack.push(2)
stack.push(3)
print(len(stack))   # 3
print(stack[0])     # 1
print(stack)        # Stack([1, 2, 3])

七、设计模式示例

7.1 单例模式

class Singleton:
    _instance = None
    
    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance
    
    def __init__(self, name):
        if not hasattr(self, 'initialized'):
            self.name = name
            self.initialized = True

# 测试
s1 = Singleton("实例1")
s2 = Singleton("实例2")
print(s1 is s2)   # True
print(s1.name)    # 实例1
print(s2.name)    # 实例1

7.2 工厂模式

class Animal(ABC):
    @abstractmethod
    def speak(self):
        pass

class Dog(Animal):
    def speak(self):
        return "汪汪汪"

class Cat(Animal):
    def speak(self):
        return "喵喵喵"

class AnimalFactory:
    @staticmethod
    def create_animal(animal_type):
        animals = {
            'dog': Dog,
            'cat': Cat
        }
        return animals[animal_type]()

# 使用工厂
factory = AnimalFactory()
dog = factory.create_animal('dog')
print(dog.speak())  # 汪汪汪

八、注意事项

最佳实践

  1. 遵循单一职责原则：一个类只做一件事
  2. 使用私有属性保护数据
  3. 优先使用组合而非继承
  4. 使用 @property 提供友好的接口
  5. 保持类和方法简短

注意

  • Python 中一切皆对象，类也是对象
  • 多继承可能导致方法解析顺序混乱
  • 魔术方法不要直接调用，使用对应的内置函数
