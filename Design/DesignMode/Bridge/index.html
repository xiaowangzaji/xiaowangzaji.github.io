桥接模式

约 1564 字大约 5 分钟

2025-09-24

定义

桥接模式将抽象部分与实现部分分离，使它们可以独立地变化。它通过组合代替继承，避免了多层子类继承导致的类爆炸问题。核心思想是"用聚合代替继承"。

结构

  • Abstraction（抽象类）：定义高层抽象接口，持有一个 Implementor 的引用。
  • RefinedAbstraction（扩展抽象类）：继承 Abstraction，扩展抽象接口。
  • Implementor（实现接口）：定义实现层的接口，不一定要和 Abstraction 完全一致。
  • ConcreteImplementor（具体实现类）：实现 Implementor 接口，提供具体的实现。

桥接模式的关键在于识别出系统中的两个独立变化维度，然后将它们分离开来。例如：形状（圆、矩形）× 颜色（红、蓝），如果不分离需要 2×2=4 个子类；分离后只需 2 个形状类 + 2 个颜色类。

UML 类图（文本描述）

Abstraction --> Implementor（组合/关联）
Abstraction <|-- RefinedAbstractionA
Abstraction <|-- RefinedAbstractionB
Implementor <|-- ConcreteImplementorA
Implementor <|-- ConcreteImplementorB
  • Abstraction 通过聚合持有 Implementor
  • RefinedAbstraction 扩展 Abstraction 的行为
  • ConcreteImplementor 提供具体的实现细节
  • 两个维度可以独立扩展，互不影响

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <memory>

// ========== 实现接口：渲染引擎 ==========
class IRenderer {
public:
    virtual ~IRenderer() = default;
    virtual void renderCircle(float x, float y, float radius) = 0;
    virtual void renderRectangle(float x1, float y1, float x2, float y2) = 0;
};

// ========== 具体实现：矢量渲染 ==========
class VectorRenderer : public IRenderer {
public:
    void renderCircle(float x, float y, float radius) override {
        std::cout << "[矢量渲染] 绘制圆形: 中心(" << x << "," << y
                  << ") 半径=" << radius << std::endl;
    }
    void renderRectangle(float x1, float y1, float x2, float y2) override {
        std::cout << "[矢量渲染] 绘制矩形: (" << x1 << "," << y1
                  << ") -> (" << x2 << "," << y2 << ")" << std::endl;
    }
};

// ========== 具体实现：光栅渲染 ==========
class RasterRenderer : public IRenderer {
public:
    void renderCircle(float x, float y, float radius) override {
        std::cout << "[光栅渲染] 光栅化圆形: 中心(" << x << "," << y
                  << ") 半径=" << radius << std::endl;
    }
    void renderRectangle(float x1, float y1, float x2, float y2) override {
        std::cout << "[光栅渲染] 光栅化矩形: (" << x1 << "," << y1
                  << ") -> (" << x2 << "," << y2 << ")" << std::endl;
    }
};

// ========== 抽象类：形状 ==========
class Shape {
protected:
    std::unique_ptr<IRenderer> renderer_;
public:
    explicit Shape(std::unique_ptr<IRenderer> renderer)
        : renderer_(std::move(renderer)) {}
    virtual ~Shape() = default;
    virtual void draw() = 0;
    virtual void resize(float factor) = 0;
};

// ========== 扩展抽象：圆形 ==========
class Circle : public Shape {
private:
    float x_, y_, radius_;
public:
    Circle(float x, float y, float r, std::unique_ptr<IRenderer> renderer)
        : Shape(std::move(renderer)), x_(x), y_(y), radius_(r) {}

    void draw() override {
        renderer_->renderCircle(x_, y_, radius_);
    }
    void resize(float factor) override {
        radius_ *= factor;
        std::cout << "  缩放后半径: " << radius_ << std::endl;
    }
};

// ========== 扩展抽象：矩形 ==========
class Rectangle : public Shape {
private:
    float x1_, y1_, x2_, y2_;
public:
    Rectangle(float x1, float y1, float x2, float y2,
              std::unique_ptr<IRenderer> renderer)
        : Shape(std::move(renderer)),
          x1_(x1), y1_(y1), x2_(x2), y2_(y2) {}

    void draw() override {
        renderer_->renderRectangle(x1_, y1_, x2_, y2_);
    }
    void resize(float factor) override {
        float cx = (x1_ + x2_) / 2, cy = (y1_ + y2_) / 2;
        x1_ = cx - (cx - x1_) * factor;
        y1_ = cy - (cy - y1_) * factor;
        x2_ = cx + (x2_ - cx) * factor;
        y2_ = cy + (y2_ - cy) * factor;
        std::cout << "  缩放后: (" << x1_ << "," << y1_
                  << ") -> (" << x2_ << "," << y2_ << ")" << std::endl;
    }
};

int main() {
    // 矢量渲染的圆形
    Circle circle1(100, 200, 50,
                   std::make_unique<VectorRenderer>());
    circle1.draw();
    circle1.resize(1.5f);
    circle1.draw();

    std::cout << "---\n";

    // 光栅渲染的矩形
    Rectangle rect(10, 20, 200, 100,
                   std::make_unique<RasterRenderer>());
    rect.draw();
    rect.resize(2.0f);
    rect.draw();

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod


# ========== 实现接口：消息平台 ==========
class MessagePlatform(ABC):
    @abstractmethod
    def send(self, msg_type: str, title: str, content: str) -> None:
        """发送消息"""
        pass


class WeChatPlatform(MessagePlatform):
    def send(self, msg_type: str, title: str, content: str) -> None:
        print(f"[微信] 发送{msg_type}消息 | 标题: {title}")
        print(f"        内容: {content}")


class EmailPlatform(MessagePlatform):
    def send(self, msg_type: str, title: str, content: str) -> None:
        print(f"[邮件] 发送{msg_type}邮件 | 主题: {title}")
        print(f"        内容: {content}")


class DingTalkPlatform(MessagePlatform):
    def send(self, msg_type: str, title: str, content: str) -> None:
        print(f"[钉钉] 发送{msg_type}消息 | 标题: {title}")
        print(f"        内容: {content}")


# ========== 抽象类：消息 ==========
class Message(ABC):
    def __init__(self, platform: MessagePlatform) -> None:
        self._platform = platform

    @abstractmethod
    def send(self, title: str, content: str) -> None:
        pass


# ========== 扩展抽象：普通消息 ==========
class CommonMessage(Message):
    def send(self, title: str, content: str) -> None:
        self._platform.send("普通", title, content)


# ========== 扩展抽象：紧急消息 ==========
class UrgentMessage(Message):
    def send(self, title: str, content: str) -> None:
        self._platform.send("紧急", title, f"【紧急】{content}")

    def notify_monitor(self, title: str) -> None:
        print(f"  → 同时监控消息「{title}」的送达状态")


# ========== 扩展抽象：特急消息 ==========
class SpecialUrgentMessage(Message):
    def send(self, title: str, content: str) -> None:
        self._platform.send("特急", title, f"【特急！】{content}")
        print("  → 触发电话通知备选通道")


if __name__ == "__main__":
    print("=== 桥接模式：消息系统 ===\n")

    # 普通消息 + 微信
    msg1 = CommonMessage(WeChatPlatform())
    msg1.send("会议通知", "明天下午3点开会")

    print()

    # 紧急消息 + 邮件
    msg2 = UrgentMessage(EmailPlatform())
    msg2.send("系统告警", "数据库主从延迟超过 5s")
    msg2.notify_monitor("系统告警")

    print()

    # 特急消息 + 钉钉
    msg3 = SpecialUrgentMessage(DingTalkPlatform())
    msg3.send("生产事故", "支付服务宕机，请立即处理")

    print("\n--- 分析 ---")
    print("抽象维度（消息类型）: 普通、紧急、特急")
    print("实现维度（发送平台）: 微信、邮件、钉钉")
    print("桥接后: 3 × 3 = 9 种组合，但只需 3 + 3 = 6 个类")
    print("若用继承: 需要 3 × 3 = 9 个类，新增平台/消息类型都要成倍增加")

适用场景

  • 系统有多个独立变化的维度，需要避免类爆炸
  • 需要在抽象和实现之间建立松耦合的关联
  • 需要在运行时动态切换实现
  • 不希望用继承导致系统复杂度急剧增加

优点

  • 解耦两个维度：抽象和实现可以独立变化
  • 避免类爆炸：M × N 的组合只需 M + N 个类
  • 运行时切换：可以在运行时动态替换实现
  • 更好的扩展性：新增维度不影响已有代码

缺点

  • 理解门槛高：需要正确识别分离的维度
  • 设计复杂度：增加了系统的抽象层次
  • 过度设计风险：如果只有一个维度变化，不需要桥接

与其他模式的关系

  • 桥接 vs 适配器：桥接是事前设计，适配器是事后补救
  • 桥接 vs 策略：策略关注算法替换，桥接关注两个维度的解耦
  • 桥接 vs 抽象工厂：抽象工厂创建一族相关对象，桥接解耦抽象和实现
  • 桥接 vs 状态：桥接的 Implementor 类似 State，但关注点不同

实际应用

  • JDBC 驱动：java.sql.Connection 是抽象，各数据库厂商的驱动是实现
  • AWT/Swing：组件（Button、TextField）与 Peer（本地平台实现）的桥接
  • 日志框架：Logger（抽象）与 Appender（实现：Console、File、Network）
  • 驱动开发：操作系统设备驱动模型中，设备类与驱动实现的桥接
  • 跨平台 UI 框架：Flutter/React Native 中，Widget 抽象与平台渲染实现的桥接
