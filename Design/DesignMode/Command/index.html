命令模式

约 1654 字大约 6 分钟

2025-09-24

定义

命令模式将一个请求封装为一个对象，从而使你可用不同的请求对客户进行参数化、对请求排队或记录请求日志，以及支持可撤销的操作。核心思想是"将方法调用封装成对象"。

结构

  • Command（命令接口）：声明执行操作的接口，通常包含 execute() 和 undo() 方法。
  • ConcreteCommand（具体命令）：实现 Command 接口，持有接收者（Receiver）的引用，调用接收者的相应操作。
  • Receiver（接收者）：知道如何实施与执行一个请求相关的操作，任何类都可能作为接收者。
  • Invoker（调用者）：要求命令执行请求，持有一个或多个命令对象。
  • Client（客户端）：创建具体命令对象并设置其接收者，然后将命令交给调用者。

命令模式的关键价值在于：命令是一个一等公民，可以被传递、存储、组合、撤销、重做、排队。

UML 类图（文本描述）

Client --> ConcreteCommand（创建命令并设置接收者）
ConcreteCommand --> Receiver（关联：持有接收者引用）
Command <|-- ConcreteCommand
Invoker --> Command（关联：持有命令引用）

执行流程: Client → 创建 ConcreteCommand → 交给 Invoker → Invoker.execute() → Command.execute() → Receiver.action()

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <vector>
#include <stack>
#include <memory>

// ========== 接收者：文本编辑器 ==========
class TextEditor {
private:
    std::string text_;
public:
    void insertText(const std::string& t) { text_ += t; }
    void deleteText(int count) {
        if (count <= static_cast<int>(text_.size()))
            text_.resize(text_.size() - count);
        else
            text_.clear();
    }
    std::string getText() const { return text_; }
    int size() const { return static_cast<int>(text_.size()); }
};

// ========== 命令接口 ==========
class Command {
public:
    virtual ~Command() = default;
    virtual void execute() = 0;
    virtual void undo() = 0;
    virtual std::string description() const = 0;
};

// ========== 具体命令：插入文本 ==========
class InsertCommand : public Command {
private:
    TextEditor* editor_;
    std::string text_;
public:
    InsertCommand(TextEditor* editor, const std::string& text)
        : editor_(editor), text_(text) {}

    void execute() override {
        std::cout << "  [插入] \"" << text_ << "\"\n";
        editor_->insertText(text_);
    }
    void undo() override {
        std::cout << "  [撤销插入] 删除 \"" << text_ << "\"\n";
        editor_->deleteText(static_cast<int>(text_.size()));
    }
    std::string description() const override { return "插入文本"; }
};

// ========== 具体命令：删除文本 ==========
class DeleteCommand : public Command {
private:
    TextEditor* editor_;
    int count_;
    std::string deleted_;  // 保存删除的内容以便撤销
public:
    DeleteCommand(TextEditor* editor, int count)
        : editor_(editor), count_(count) {}

    void execute() override {
        int start = editor_->size() - count_;
        if (start < 0) start = 0;
        deleted_ = editor_->getText().substr(start);
        std::cout << "  [删除] " << count_ << " 个字符: \"" << deleted_ << "\"\n";
        editor_->deleteText(count_);
    }
    void undo() override {
        std::cout << "  [撤销删除] 恢复 \"" << deleted_ << "\"\n";
        editor_->insertText(deleted_);
    }
    std::string description() const override { return "删除文本"; }
};

// ========== 调用者：命令历史（支持撤销/重做） ==========
class CommandHistory {
private:
    std::stack<std::shared_ptr<Command>> undoStack_;
    std::stack<std::shared_ptr<Command>> redoStack_;
public:
    void executeCommand(std::shared_ptr<Command> cmd) {
        cmd->execute();
        undoStack_.push(cmd);
        // 清空重做栈
        while (!redoStack_.empty()) redoStack_.pop();
    }
    void undo() {
        if (undoStack_.empty()) {
            std::cout << "  无可撤销操作\n";
            return;
        }
        auto cmd = undoStack_.top();
        undoStack_.pop();
        cmd->undo();
        redoStack_.push(cmd);
    }
    void redo() {
        if (redoStack_.empty()) {
            std::cout << "  无可重做操作\n";
            return;
        }
        auto cmd = redoStack_.top();
        redoStack_.pop();
        cmd->execute();
        undoStack_.push(cmd);
    }
};

int main() {
    TextEditor editor;
    CommandHistory history;

    auto cmd1 = std::make_shared<InsertCommand>(&editor, "Hello ");
    auto cmd2 = std::make_shared<InsertCommand>(&editor, "World");
    auto cmd3 = std::make_shared<DeleteCommand>(&editor, 5);

    std::cout << "=== 编辑操作 ===\n";
    history.executeCommand(cmd1);
    history.executeCommand(cmd2);
    std::cout << "当前文本: \"" << editor.getText() << "\"\n";

    std::cout << "\n=== 撤销 ===\n";
    history.undo();
    std::cout << "当前文本: \"" << editor.getText() << "\"\n";
    history.undo();
    std::cout << "当前文本: \"" << editor.getText() << "\"\n";

    std::cout << "\n=== 重做 ===\n";
    history.redo();
    std::cout << "当前文本: \"" << editor.getText() << "\"\n";

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from typing import List


# ========== 接收者：智能家居系统 ==========
class Light:
    def __init__(self, name: str) -> None:
        self._name = name
        self._on = False

    def turn_on(self) -> None:
        self._on = True
        print(f"  💡 {self._name} 灯已打开")

    def turn_off(self) -> None:
        self._on = False
        print(f"  💡 {self._name} 灯已关闭")

    @property
    def is_on(self) -> bool:
        return self._on


class AirConditioner:
    def __init__(self, name: str) -> None:
        self._name = name
        self._temperature = 26

    def set_temperature(self, temp: int) -> None:
        self._temperature = temp
        print(f"  🌡 {self._name} 温度设为 {temp}°C")

    def turn_on(self) -> None:
        print(f"  ❄ {self._name} 空调已打开")

    def turn_off(self) -> None:
        print(f"  ❄ {self._name} 空调已关闭")


# ========== 命令接口 ==========
class Command(ABC):
    @abstractmethod
    def execute(self) -> None: ...

    @abstractmethod
    def undo(self) -> None: ...

    @abstractmethod
    def description(self) -> str: ...


# ========== 具体命令 ==========
class LightOnCommand(Command):
    def __init__(self, light: Light) -> None:
        self._light = light

    def execute(self) -> None:
        self._light.turn_on()

    def undo(self) -> None:
        self._light.turn_off()

    def description(self) -> str:
        return f"打开{self._light._name}灯"


class LightOffCommand(Command):
    def __init__(self, light: Light) -> None:
        self._light = light

    def execute(self) -> None:
        self._light.turn_off()

    def undo(self) -> None:
        self._light.turn_on()

    def description(self) -> str:
        return f"关闭{self._light._name}灯"


class ACTemperatureCommand(Command):
    def __init__(self, ac: AirConditioner, temperature: int) -> None:
        self._ac = ac
        self._temperature = temperature
        self._prev_temp = ac._temperature

    def execute(self) -> None:
        self._prev_temp = self._ac._temperature
        self._ac.set_temperature(self._temperature)

    def undo(self) -> None:
        self._ac.set_temperature(self._prev_temp)

    def description(self) -> str:
        return f"设{self._ac._name}温度为{self._temperature}°C"


# ========== 宏命令（组合命令） ==========
class MacroCommand(Command):
    def __init__(self, commands: List[Command]) -> None:
        self._commands = commands

    def execute(self) -> None:
        for cmd in self._commands:
            cmd.execute()

    def undo(self) -> None:
        for cmd in reversed(self._commands):
            cmd.undo()

    def description(self) -> str:
        return "宏命令: " + ", ".join(c.description() for c in self._commands)


# ========== 调用者：遥控器 ==========
class RemoteControl:
    def __init__(self) -> None:
        self._commands: dict[str, Command] = {}
        self._history: List[Command] = []

    def set_command(self, button: str, command: Command) -> None:
        self._commands[button] = command

    def press_button(self, button: str) -> None:
        if button not in self._commands:
            print(f"  按钮 {button} 未设置命令")
            return
        cmd = self._commands[button]
        print(f"\n执行: {cmd.description()}")
        cmd.execute()
        self._history.append(cmd)

    def undo_last(self) -> None:
        if not self._history:
            print("  无操作可撤销")
            return
        cmd = self._history.pop()
        print(f"\n撤销: {cmd.description()}")
        cmd.undo()


if __name__ == "__main__":
    # 接收者
    living_light = Light("客厅")
    bedroom_light = Light("卧室")
    ac = AirConditioner("客厅空调")

    # 遥控器
    remote = RemoteControl()

    # 设置按钮
    remote.set_command("1", LightOnCommand(living_light))
    remote.set_command("2", LightOffCommand(living_light))
    remote.set_command("3", LightOnCommand(bedroom_light))
    remote.set_command("4", ACTemperatureCommand(ac, 22))

    # 宏命令：回家模式
    home_mode = MacroCommand([
        LightOnCommand(living_light),
        LightOnCommand(bedroom_light),
        ACTemperatureCommand(ac, 24),
    ])
    remote.set_command("0", home_mode)

    # 操作
    remote.press_button("0")  # 回家模式
    remote.press_button("2")  # 关客厅灯
    remote.undo_last()        # 撤销

适用场景

  • 需要支持撤销/重做操作（编辑器、游戏）
  • 需要将操作请求排队、记录日志或持久化
  • 需要将调用者与接收者解耦
  • 需要实现宏命令（一组操作的组合）
  • GUI 按钮、菜单项的命令绑定

优点

  • 开闭原则：新增命令无需修改调用者代码
  • 撤销/重做：天然支持，只需记录命令历史
  • 解耦：调用者无需知道接收者的存在
  • 组合：可以组合多个命令形成宏命令
  • 日志/持久化：命令对象可以序列化和存储

缺点

  • 类数量增加：每个操作都需要一个命令类
  • 复杂度提升：简单场景下可能过度设计

与其他模式的关系

  • 命令 vs 策略：命令封装请求（可撤销），策略封装算法
  • 命令 vs 备忘录：备忘录保存状态用于恢复，命令通过反向操作撤销
  • 命令 vs 中介者：命令是一对一委托，中介者是多对多协调
  • 命令 vs Memento：命令模式可以结合备忘录模式实现更强大的撤销

实际应用

  • 编辑器：所有编辑操作都封装为命令（Ctrl+Z/Ctrl+Y）
  • GUI 框架：按钮点击、菜单项、快捷键的命令绑定
  • 事务系统：数据库事务本质上是一组命令
  • 游戏开发：输入命令队列（回放系统基于命令重放）
  • 消息队列：RabbitMQ/Kafka 中的消息就是命令
  • Redux：前端状态管理中的 Action/Reducer 就是命令模式的变体
