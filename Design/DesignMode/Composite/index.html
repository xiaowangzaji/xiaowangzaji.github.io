组合模式

约 1729 字大约 6 分钟

2025-09-24

定义

组合模式将对象组合成树形结构以表示"部分-整体"的层次结构。组合模式使得客户端对单个对象和组合对象的使用具有一致性，即可以用相同的方式处理个体对象和组合对象。

结构

  • Component（组件接口）：为所有对象声明公共接口，定义默认行为，提供管理子组件的接口（增、删、获取）。
  • Leaf（叶子节点）：表示树形结构中的叶子对象，没有子节点，实现 Component 接口的基本操作。
  • Composite（组合节点）：表示有子节点的容器对象，实现了管理子组件的方法，并将操作委派给子组件。
  • Client（客户端）：通过 Component 接口操作树中的对象，无需区分叶子和容器。

组合模式有两种安全实现方式：透明组合（Component 包含所有方法，叶子节点对容器方法返回空或抛异常）和安全组合（叶子节点没有管理子节点的方法，但客户端需要区分类型）。

UML 类图（文本描述）

Component（定义公共接口和子节点管理方法）
    |-- Leaf（叶子节点，无子节点）
    |-- Composite（容器节点，包含子节点列表）
            --> Component（聚合：持有子节点引用列表）

Client --> Component（统一操作，无需区分叶子/容器）

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <vector>
#include <memory>
#include <numeric>

// ========== 组件接口 ==========
class FileSystemComponent {
protected:
    std::string name_;
public:
    explicit FileSystemComponent(const std::string& name) : name_(name) {}
    virtual ~FileSystemComponent() = default;

    virtual void showInfo(int indent = 0) = 0;
    virtual long getSize() const = 0;
    virtual void add(std::shared_ptr<FileSystemComponent>) {}
    virtual void remove(std::shared_ptr<FileSystemComponent>) {}
    virtual std::shared_ptr<FileSystemComponent> getChild(int) { return nullptr; }
    virtual bool isComposite() const { return false; }
};

// ========== 叶子节点：文件 ==========
class File : public FileSystemComponent {
private:
    long size_;
public:
    File(const std::string& name, long size)
        : FileSystemComponent(name), size_(size) {}

    void showInfo(int indent = 0) override {
        std::string prefix(indent * 2, ' ');
        std::cout << prefix << "📄 " << name_ << " (" << size_ << " bytes)"
                  << std::endl;
    }
    long getSize() const override { return size_; }
    bool isComposite() const override { return false; }
};

// ========== 组合节点：文件夹 ==========
class Directory : public FileSystemComponent {
private:
    std::vector<std::shared_ptr<FileSystemComponent>> children_;
public:
    explicit Directory(const std::string& name)
        : FileSystemComponent(name) {}

    void add(std::shared_ptr<FileSystemComponent> child) override {
        children_.push_back(child);
    }
    void remove(std::shared_ptr<FileSystemComponent> child) override {
        children_.erase(
            std::remove(children_.begin(), children_.end(), child),
            children_.end()
        );
    }
    std::shared_ptr<FileSystemComponent> getChild(int index) override {
        return children_.at(index);
    }

    void showInfo(int indent = 0) override {
        std::string prefix(indent * 2, ' ');
        std::cout << prefix << "📁 " << name_ << "/" << std::endl;
        for (auto& child : children_) {
            child->showInfo(indent + 1);
        }
    }
    long getSize() const override {
        return std::accumulate(
            children_.begin(), children_.end(), 0L,
            [](long sum, const auto& child) { return sum + child->getSize(); }
        );
    }
    bool isComposite() const override { return true; }
};

// ========== 实际场景：UI 组件树 ==========
class UIComponent {
protected:
    std::string tag_;
public:
    explicit UIComponent(const std::string& tag) : tag_(tag) {}
    virtual ~UIComponent() = default;
    virtual void render(int depth = 0) = 0;
    virtual void add(std::shared_ptr<UIComponent>) {}
    virtual void remove(std::shared_ptr<UIComponent>) {}
};

class Button : public UIComponent {
private:
    std::string label_;
public:
    Button(const std::string& tag, const std::string& label)
        : UIComponent(tag), label_(label) {}
    void render(int depth = 0) override {
        std::string indent(depth * 2, ' ');
        std::cout << indent << "<" << tag_ << ">" << label_ << "</" << tag_ << ">"
                  << std::endl;
    }
};

class Panel : public UIComponent {
private:
    std::vector<std::shared_ptr<UIComponent>> children_;
public:
    explicit Panel(const std::string& tag) : UIComponent(tag) {}
    void add(std::shared_ptr<UIComponent> child) override {
        children_.push_back(child);
    }
    void render(int depth = 0) override {
        std::string indent(depth * 2, ' ');
        std::cout << indent << "<" << tag_ << ">" << std::endl;
        for (auto& c : children_) c->render(depth + 1);
        std::cout << indent << "</" << tag_ << ">" << std::endl;
    }
};

int main() {
    // 文件系统示例
    auto root = std::make_shared<Directory>("/home");
    auto docs = std::make_shared<Directory>("docs");
    auto src = std::make_shared<Directory>("src");

    root->add(docs);
    root->add(src);
    root->add(std::make_shared<File>("README.md", 2048));
    docs->add(std::make_shared<File>("design.md", 5120));
    docs->add(std::make_shared<File>("notes.md", 3072));
    src->add(std::make_shared<File>("main.cpp", 8192));
    src->add(std::make_shared<File>("utils.cpp", 4096));

    root->showInfo();
    std::cout << "\n总大小: " << root->getSize() << " bytes\n";

    // UI 组件树
    std::cout << "\n=== UI 组件树 ===\n";
    auto form = std::make_shared<Panel>("form");
    form->add(std::make_shared<Panel>("div"));
    form->add(std::make_shared<Button>("button", "提交"));
    form->add(std::make_shared<Button>("button", "取消"));
    form->render();

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from typing import List


# ========== 组件接口 ==========
class Graphic(ABC):
    @abstractmethod
    def draw(self) -> None:
        pass

    @abstractmethod
    def move(self, x: int, y: int) -> None:
        pass


# ========== 叶子节点 ==========
class Dot(Graphic):
    def __init__(self, x: int, y: int) -> None:
        self._x, self._y = x, y

    def draw(self) -> None:
        print(f"  · 点在 ({self._x}, {self._y})")

    def move(self, x: int, y: int) -> None:
        self._x += x
        self._y += y


class Circle(Dot):
    def __init__(self, x: int, y: int, radius: float) -> None:
        super().__init__(x, y)
        self._radius = radius

    def draw(self) -> None:
        print(f"  ○ 圆形 中心({self._x}, {self._y}) 半径={self._radius}")


class Rectangle(Graphic):
    def __init__(self, x: int, y: int, w: int, h: int) -> None:
        self._x, self._y, self._w, self._h = x, y, w, h

    def draw(self) -> None:
        print(f"  □ 矩形 左上({self._x}, {self._y}) 尺寸={self._w}×{self._h}")

    def move(self, x: int, y: int) -> None:
        self._x += x
        self._y += y


# ========== 组合节点 ==========
class CompoundGraphic(Graphic):
    def __init__(self) -> None:
        self._children: List[Graphic] = []

    def add(self, child: Graphic) -> None:
        self._children.append(child)

    def remove(self, child: Graphic) -> None:
        self._children.remove(child)

    def draw(self) -> None:
        print(f"  [组合图形，包含 {len(self._children)} 个元素]")
        for child in self._children:
            child.draw()

    def move(self, x: int, y: int) -> None:
        for child in self._children:
            child.move(x, y)


# ========== 实际应用：组织架构图 ==========
class Employee:
    def __init__(self, name: str, title: str, salary: float) -> None:
        self._name = name
        self._title = title
        self._salary = salary
        self._subordinates: List["Employee"] = []

    def add(self, emp: "Employee") -> None:
        self._subordinates.append(emp)

    def remove(self, emp: "Employee") -> None:
        self._subordinates.remove(emp)

    def get_total_salary(self) -> float:
        """递归计算整个团队的总薪资"""
        return self._salary + sum(
            sub.get_total_salary() for sub in self._subordinates
        )

    def display(self, indent: int = 0) -> None:
        prefix = "  " * indent
        print(f"{prefix}{self._name} [{self._title}] ¥{self._salary:,.0f}")
        for sub in self._subordinates:
            sub.display(indent + 1)


if __name__ == "__main__":
    print("=== 组合模式：图形编辑器 ===\n")
    all_graphics = CompoundGraphic()
    all_graphics.add(Dot(1, 2))
    all_graphics.add(Circle(5, 3, 10))

    group = CompoundGraphic()
    group.add(Rectangle(10, 10, 50, 30))
    group.add(Circle(20, 20, 5))
    all_graphics.add(group)

    all_graphics.draw()

    print("\n移动整个图形 (+10, +20)...")
    all_graphics.move(10, 20)
    all_graphics.draw()

    print("\n=== 组合模式：组织架构 ===\n")
    ceo = Employee("张总", "CEO", 100000)
    cto = Employee("李CTO", "CTO", 80000)
    cfo = Employee("王CFO", "CFO", 75000)
    dev1 = Employee("赵开发", "高级工程师", 40000)
    dev2 = Employee("钱开发", "初级工程师", 25000)

    ceo.add(cto)
    ceo.add(cfo)
    cto.add(dev1)
    cto.add(dev2)

    ceo.display()
    print(f"\n公司总人力成本: ¥{ceo.get_total_salary():,.0f}")

适用场景

  • 需要表示对象的部分-整体层次结构
  • 希望客户端统一处理单个对象和组合对象
  • 树形结构：文件系统、UI 组件树、组织架构、菜单系统
  • XML/HTML 文档的 DOM 树

优点

  • 统一接口：客户端无需区分叶子和容器
  • 简化客户端代码：递归操作天然适合树形结构
  • 易于扩展：新增叶子或容器类型不影响现有代码
  • 通用性：任何树形结构都可以使用

缺点

  • 类型安全：透明组合中叶子节点的容器方法可能无意义
  • 过度通用：Component 接口可能变得过于庞大
  • 遍历限制：某些操作需要特定类型的节点，通用接口不够精确

与其他模式的关系

  • 组合 vs 责任链：组合是树形委派，责任链是线性委派
  • 组合 vs 迭代器：组合常与迭代器配合遍历树节点
  • 组合 vs 装饰器：两者都是递归结构，但组合构建层次，装饰器增强行为
  • 组合 vs 访问者：对组合树结构的操作常使用访问者模式

实际应用

  • DOM API：HTML 元素树是组合模式的经典应用
  • 文件系统：目录/文件的层次结构
  • Swing/AWT：Container 可以包含 Component（包括其他 Container）
  • React/Vue：组件树就是组合模式，父组件包含子组件
  • JSON/XML 解析：文档对象模型（DOM）树
  • 菜单系统：菜单项和子菜单的组合
