外观模式

约 1768 字大约 6 分钟

2025-09-24

定义

外观模式为子系统中的一组接口提供一个统一的高层接口，使得子系统更容易使用。它不添加新功能，而是将复杂的子系统接口简化为简单的门面，是迪米特法则（最少知识原则）的典型应用。

结构

  • Facade（外观）：知道哪些子系统类负责处理请求，将客户端请求委派给适当的子系统对象。
  • Subsystem（子系统类）：实现子系统的功能，处理 Facade 指派的任务。子系统不知道 Facade 的存在，对 Facade 一无所知。
  • Client（客户端）：通过 Facade 与子系统交互，不直接访问子系统类。

外观模式与代理模式的区别：代理模式是一对一包装同一个对象，外观模式是一对多简化整个子系统。

UML 类图（文本描述）

Client --> Facade（依赖）
Facade --> SubsystemA（组合/关联）
Facade --> SubsystemB（组合/关联）
Facade --> SubsystemC（组合/关联）
  • Facade 持有多个子系统对象的引用
  • 客户端只与 Facade 交互，不需要了解子系统内部结构
  • 子系统类之间可以互相调用，但它们不知道 Facade 的存在

代码实现（C++ 示例）

#include <iostream>
#include <string>

// ========== 子系统：CPU ==========
class CPU {
public:
    void freeze() {
        std::cout << "  [CPU] 冻结处理器" << std::endl;
    }
    void jump(long position) {
        std::cout << "  [CPU] 跳转到地址: " << position << std::endl;
    }
    void execute() {
        std::cout << "  [CPU] 执行指令" << std::endl;
    }
};

// ========== 子系统：内存 ==========
class Memory {
public:
    void load(long position, const std::string& data) {
        std::cout << "  [Memory] 加载数据到地址 " << position
                  << ": " << data << std::endl;
    }
};

// ========== 子系统：硬盘 ==========
class HardDrive {
public:
    std::string read(long lba, int size) {
        std::string data = "BOOT_SECTOR_DATA";
        std::cout << "  [HardDrive] 读取扇区 LBA=" << lba
                  << ", 大小=" << size << std::endl;
        return data;
    }
};

// ========== 外观：计算机启动 ==========
class ComputerFacade {
private:
    CPU cpu_;
    Memory memory_;
    HardDrive hardDrive_;

    static constexpr long BOOT_ADDRESS = 0x7C00;
    static constexpr int SECTOR_SIZE = 512;

public:
    void startComputer() {
        std::cout << "=== 启动计算机 ===" << std::endl;
        cpu_.freeze();
        memory_.load(BOOT_ADDRESS, hardDrive_.read(0, SECTOR_SIZE));
        cpu_.jump(BOOT_ADDRESS);
        cpu_.execute();
        std::cout << "=== 计算机启动完成 ===" << std::endl;
    }
};

// ========== 另一个例子：家庭影院系统 ==========
class Projector {
public:
    void on() { std::cout << "  [投影仪] 打开" << std::endl; }
    void setInput(const std::string& source) {
        std::cout << "  [投影仪] 切换到 " << source << std::endl;
    }
    void off() { std::cout << "  [投影仪] 关闭" << std::endl; }
};

class Amplifier {
public:
    void on() { std::cout << "  [功放] 打开" << std::endl; }
    void setVolume(int level) {
        std::cout << "  [功放] 设置音量: " << level << std::endl;
    }
    void setInput(const std::string& source) {
        std::cout << "  [功放] 切换到 " << source << std::endl;
    }
    void off() { std::cout << "  [功放] 关闭" << std::endl; }
};

class DVDPlayer {
public:
    void on() { std::cout << "  [DVD] 打开" << std::endl; }
    void play(const std::string& movie) {
        std::cout << "  [DVD] 播放: " << movie << std::endl;
    }
    void stop() { std::cout << "  [DVD] 停止" << std::endl; }
    void off() { std::cout << "  [DVD] 关闭" << std::endl; }
};

class Lights {
public:
    void dim(int level) {
        std::cout << "  [灯光] 调暗到 " << level << "%" << std::endl;
    }
    void on() { std::cout << "  [灯光] 打开" << std::endl; }
};

class PopcornPopper {
public:
    void on() { std::cout << "  [爆米花机] 打开" << std::endl; }
    void pop() { std::cout << "  [爆米花机] 制作爆米花" << std::endl; }
    void off() { std::cout << "  [爆米花机] 关闭" << std::endl; }
};

// 家庭影院外观
class HomeTheaterFacade {
private:
    Projector& projector_;
    Amplifier& amplifier_;
    DVDPlayer& dvdPlayer_;
    Lights& lights_;
    PopcornPopper& popcornPopper_;

public:
    HomeTheaterFacade(Projector& p, Amplifier& a, DVDPlayer& d,
                      Lights& l, PopcornPopper& pp)
        : projector_(p), amplifier_(a), dvdPlayer_(d),
          lights_(l), popcornPopper_(pp) {}

    void watchMovie(const std::string& movie) {
        std::cout << "\n=== 准备观影 ===" << std::endl;
        popcornPopper_.on();
        popcornPopper_.pop();
        lights_.dim(10);
        projector_.on();
        projector_.setInput("DVD");
        amplifier_.on();
        amplifier_.setInput("DVD");
        amplifier_.setVolume(5);
        dvdPlayer_.on();
        dvdPlayer_.play(movie);
        std::cout << "=== 电影开始 ===\n" << std::endl;
    }

    void endMovie() {
        std::cout << "\n=== 结束观影 ===" << std::endl;
        dvdPlayer_.stop();
        dvdPlayer_.off();
        amplifier_.off();
        projector_.off();
        lights_.on();
        popcornPopper_.off();
        std::cout << "=== 影院关闭 ===\n" << std::endl;
    }
};

int main() {
    // 计算机启动
    ComputerFacade computer;
    computer.startComputer();

    // 家庭影院
    Projector projector;
    Amplifier amplifier;
    DVDPlayer dvd;
    Lights lights;
    PopcornPopper popcorn;

    HomeTheaterFacade theater(projector, amplifier, dvd, lights, popcorn);
    theater.watchMovie("《星际穿越》");
    theater.endMovie();

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod


# ========== 子系统：订单服务 ==========
class OrderService:
    def create_order(self, user_id: int, items: list) -> str:
        order_id = f"ORD-{user_id}-{len(items)}"
        print(f"  [订单服务] 创建订单: {order_id}")
        return order_id

    def get_order_status(self, order_id: str) -> str:
        print(f"  [订单服务] 查询订单状态: {order_id}")
        return "已确认"


# ========== 子系统：库存服务 ==========
class InventoryService:
    def check_stock(self, item_id: str) -> int:
        print(f"  [库存服务] 检查库存: {item_id}")
        return 100  # 假设有库存

    def reserve_stock(self, item_id: str, quantity: int) -> bool:
        print(f"  [库存服务] 预留库存: {item_id} × {quantity}")
        return True


# ========== 子系统：支付服务 ==========
class PaymentService:
    def process_payment(self, order_id: str, amount: float) -> bool:
        print(f"  [支付服务] 处理支付: {order_id} 金额: ¥{amount:.2f}")
        return True


# ========== 子系统：物流服务 ==========
class ShippingService:
    def arrange_shipping(self, order_id: str, address: str) -> str:
        tracking_no = f"SF-{order_id}"
        print(f"  [物流服务] 安排发货: {order_id} -> {address}")
        print(f"  [物流服务] 运单号: {tracking_no}")
        return tracking_no


# ========== 子系统：通知服务 ==========
class NotificationService:
    def send_confirmation(self, user_id: int, order_id: str) -> None:
        print(f"  [通知服务] 发送订单确认给用户 {user_id}: {order_id}")


# ========== 外观：下单流程 ==========
class OrderFacade:
    def __init__(self) -> None:
        self.order_svc = OrderService()
        self.inventory_svc = InventoryService()
        self.payment_svc = PaymentService()
        self.shipping_svc = ShippingService()
        self.notification_svc = NotificationService()

    def place_order(
        self, user_id: int, items: list, amount: float, address: str
    ) -> dict:
        """一键下单：封装所有子系统调用"""
        print(">>> 开始下单流程")

        # 1. 检查库存
        for item_id in items:
            stock = self.inventory_svc.check_stock(item_id)
            if stock <= 0:
                raise ValueError(f"库存不足: {item_id}")

        # 2. 创建订单
        order_id = self.order_svc.create_order(user_id, items)

        # 3. 预留库存
        for item_id in items:
            self.inventory_svc.reserve_stock(item_id, 1)

        # 4. 处理支付
        if not self.payment_svc.process_payment(order_id, amount):
            raise RuntimeError("支付失败")

        # 5. 安排发货
        tracking = self.shipping_svc.arrange_shipping(order_id, address)

        # 6. 发送通知
        self.notification_svc.send_confirmation(user_id, order_id)

        print(">>> 下单完成")
        return {
            "order_id": order_id,
            "tracking": tracking,
            "status": "已发货",
        }


if __name__ == "__main__":
    facade = OrderFacade()
    result = facade.place_order(
        user_id=1001,
        items=["ITEM-001", "ITEM-002"],
        amount=299.90,
        address="北京市朝阳区XX路XX号",
    )
    print(f"\n订单结果: {result}")

    print("\n--- 外观模式总结 ---")
    print("客户端只需调用 place_order()，无需知道:")
    print("  - 库存检查的逻辑")
    print("  - 订单创建的格式")
    print("  - 支付服务的接口")
    print("  - 物流和通知的调用顺序")

适用场景

  • 需要为复杂子系统提供一个简单的入口点
  • 客户端与多个子系统之间存在大量依赖，需要解耦
  • 需要对子系统进行分层，每层用外观作为入口
  • 遗留系统重构时，用外观封装旧代码的复杂接口

优点

  • 简化接口：客户端无需了解子系统内部细节
  • 松耦合：客户端与子系统解耦，子系统变化不影响客户端
  • 更好的层次划分：可以分层设计外观，形成架构层次
  • 遵循迪米特法则：减少对象之间的依赖关系

缺点

  • 外观可能变得臃肿：随着子系统增多，Facade 类可能承担过多职责
  • 灵活性降低：客户端无法绕过外观直接访问子系统（可以通过保留子系统访问来缓解）
  • 违反开闭原则：新增子系统时可能需要修改 Facade

与其他模式的关系

  • 外观 vs 适配器：外观简化接口，适配器转换接口
  • 外观 vs 中介者：外观单向简化访问，中介者双向协调交互
  • 外观 vs 单例：Facade 通常设计为单例
  • 外观 vs 抽象工厂：抽象工厂创建对象，外观提供访问入口

实际应用

  • jQuery：$() 就是 DOM API 的外观，简化了原生 JS 操作
  • Spring Boot：@SpringBootApplication 是众多 Spring 配置的外观
  • 日志框架：SLF4J 是各种日志实现的外观门面
  • Python logging 模块：logging.info() 等是底层 Handler/Formatter 的外观
  • 数据库 ORM：SQLAlchemy/Django ORM 是 SQL 查询的外观
  • HTTP 客户端：requests.get() 是 urllib/ssl/socket 的外观
