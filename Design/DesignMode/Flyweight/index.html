享元模式

约 1809 字大约 6 分钟

2025-09-24

定义

享元模式运用共享技术来有效地支持大量细粒度对象的复用。它通过共享已经存在的对象来大幅减少需要创建的对象数量，从而降低内存消耗和提高性能。享元模式的核心是区分对象的内在状态（可共享）和外在状态（不可共享）。

结构

  • Flyweight（享元接口）：定义一个接口，享元通过它可以接受并作用于外在状态。
  • ConcreteFlyweight（具体享元）：实现享元接口，存储内在状态（不可变、可共享的部分）。
  • UnsharedConcreteFlyweight（非共享具体享元）：并非所有享元子类都需要共享。
  • FlyweightFactory（享元工厂）：创建并管理享元对象，确保合理共享（通常使用缓存/池）。
  • Client（客户端）：维持对享元的引用，计算或存储外在状态。

享元模式的关键在于正确区分哪些状态可以共享（内在状态），哪些状态必须由客户端提供（外在状态）。

UML 类图（文本描述）

Flyweight <|-- ConcreteFlyweight
FlyweightFactory --> Flyweight（关联：缓存享元对象）
Client --> Flyweight（依赖）
Client --> FlyweightFactory（依赖：获取享元实例）
  • FlyweightFactory 内部维护一个享元对象池（通常是 HashMap）
  • 客户端通过 Factory 获取享元，如果已存在则返回缓存实例
  • 客户端在调用享元方法时传入外在状态参数

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <unordered_map>
#include <memory>
#include <vector>

// ========== 享元接口：字符 ==========
struct CharacterState {
    char font;        // 字体（内在状态：可共享）
    int fontSize;     // 字号（内在状态：可共享）
    bool bold;        // 是否加粗（内在状态：可共享）
    bool italic;      // 是否斜体（内在状态：可共享）
    std::string color;// 颜色（内在状态：可共享）

    bool operator==(const CharacterState& other) const {
        return font == other.font && fontSize == other.fontSize &&
               bold == other.bold && italic == other.italic &&
               color == other.color;
    }
};

// 哈希函数
namespace std {
    template<>
    struct hash<CharacterState> {
        size_t operator()(const CharacterState& s) const {
            size_t h = 0;
            h ^= std::hash<char>{}(s.font) + 0x9e3779b9 + (h << 6) + (h >> 2);
            h ^= std::hash<int>{}(s.fontSize) + 0x9e3779b9 + (h << 6) + (h >> 2);
            h ^= std::hash<bool>{}(s.bold) + 0x9e3779b9 + (h << 6) + (h >> 2);
            h ^= std::hash<bool>{}(s.italic) + 0x9e3779b9 + (h << 6) + (h >> 2);
            h ^= std::hash<std::string>{}(s.color) + 0x9e3779b9 + (h << 6) + (h >> 2);
            return h;
        }
    };
}

class CharacterFlyweight {
private:
    CharacterState intrinsic_;  // 内在状态（共享）
public:
    explicit CharacterFlyweight(const CharacterState& state)
        : intrinsic_(state) {}

    void render(char ch, int x, int y) const {
        // 外在状态：字符内容、位置
        std::cout << "  渲染 '" << ch << "' @ (" << x << "," << y
                  << ") | 字体=" << intrinsic_.font
                  << " 大小=" << intrinsic_.fontSize
                  << " 颜色=" << intrinsic_.color
                  << (intrinsic_.bold ? " 加粗" : "")
                  << (intrinsic_.italic ? " 斜体" : "")
                  << std::endl;
    }
};

// ========== 享元工厂 ==========
class FontFactory {
private:
    std::unordered_map<CharacterState, std::shared_ptr<CharacterFlyweight>> pool_;

public:
    std::shared_ptr<CharacterFlyweight> getFont(const CharacterState& state) {
        auto it = pool_.find(state);
        if (it != pool_.end()) {
            std::cout << "  [享元工厂] 复用已有字体样式\n";
            return it->second;
        }
        std::cout << "  [享元工厂] 创建新字体样式\n";
        auto fw = std::make_shared<CharacterFlyweight>(state);
        pool_[state] = fw;
        return fw;
    }

    size_t getPoolSize() const { return pool_.size(); }
};

int main() {
    FontFactory factory;
    std::vector<std::pair<std::shared_ptr<CharacterFlyweight>, std::pair<char, int>>> document;

    // 模拟文档渲染：大量相同样式的字符
    CharacterState titleStyle{'A', 24, true, false, "#1a1a2e"};
    CharacterState bodyStyle{'S', 12, false, false, "#333333"};
    CharacterState emphasisStyle{'S', 14, true, true, "#e94560"};

    // 标题行（相同样式复用）
    std::string title = "DESIGN PATTERNS";
    for (size_t i = 0; i < title.size(); ++i) {
        auto font = factory.getFont(titleStyle);
        document.emplace_back(font, std::make_pair(title[i], 100 + static_cast<int>(i) * 20));
    }

    // 正文（相同样式复用）
    std::string body = "Flyweight pattern reduces memory usage by sharing objects.";
    for (size_t i = 0; i < body.size(); ++i) {
        auto font = factory.getFont(bodyStyle);
        document.emplace_back(font, std::make_pair(body[i], static_cast<int>(i) * 10));
    }

    // 渲染
    std::cout << "=== 渲染文档 ===\n";
    for (auto& [font, charPos] : document) {
        font->render(charPos.first, charPos.second, 0);
    }

    std::cout << "\n文档字符数: " << document.size()
              << ", 享元对象数: " << factory.getPoolSize() << std::endl;

    return 0;
}

代码实现（Python 示例）

from dataclasses import dataclass
from typing import Dict, Tuple


# ========== 内在状态：咖啡种类（可共享） ==========
@dataclass(frozen=True)
class CoffeeFlavor:
    """享元对象：咖啡口味是内在状态，可以共享"""
    name: str

    def serve(self, table: int) -> None:
        print(f"  上菜: {self.name} → {table}号桌")


# ========== 享元工厂 ==========
class CoffeeFlavorFactory:
    _flavors: Dict[str, CoffeeFlavor] = {}

    @classmethod
    def get_flavor(cls, name: str) -> CoffeeFlavor:
        if name not in cls._flavors:
            print(f"  [工厂] 创建新口味: {name}")
            cls._flavors[name] = CoffeeFlavor(name)
        else:
            print(f"  [工厂] 复用已有口味: {name}")
        return cls._flavors[name]

    @classmethod
    def total_flavors(cls) -> int:
        return len(cls._flavors)


# ========== 外在状态：订单 ==========
@dataclass
class CoffeeOrderContext:
    """订单上下文：桌号和数量是外在状态，不可共享"""
    table_number: int
    quantity: int


# ========== 实际场景：树木渲染（游戏） ==========
@dataclass(frozen=True)
class TreeType:
    """树的类型是内在状态（可共享）"""
    name: str
    color: str
    texture: str

    def draw(self, x: int, y: int) -> None:
        print(f"  🌲 [{self.name}] ({x}, {y}) | 颜色={self.color} 纹理={self.texture}")


class TreeFactory:
    _types: Dict[str, TreeType] = {}

    @classmethod
    def get_tree_type(cls, name: str, color: str, texture: str) -> TreeType:
        if name not in cls._types:
            cls._types[name] = TreeType(name, color, texture)
        return cls._types[name]


class Tree:
    """每棵树的位置是外在状态（不可共享）"""
    def __init__(self, x: int, y: int, tree_type: TreeType) -> None:
        self._x = x
        self._y = y
        self._type = tree_type

    def draw(self) -> None:
        self._type.draw(self._x, self._y)


class Forest:
    def __init__(self) -> None:
        self._trees: list[Tree] = []

    def plant_tree(self, x: int, y: int, name: str, color: str, texture: str) -> None:
        tree_type = TreeFactory.get_tree_type(name, color, texture)
        self._trees.append(Tree(x, y, tree_type))

    def draw(self) -> None:
        for tree in self._trees:
            tree.draw()

    @classmethod
    def report(cls) -> None:
        total = sum(len(trees._trees) for trees in [Forest()])  # just example
        print(f"\n树木类型数: {len(TreeFactory._types)}")
        print("（如果每棵树都单独存储类型信息，内存消耗将增加数十倍）")


if __name__ == "__main__":
    print("=== 享元模式：咖啡店 ===\n")

    factory = CoffeeFlavorFactory

    orders = [
        (factory.get_flavor("卡布奇诺"), CoffeeOrderContext(1, 2)),
        (factory.get_flavor("拿铁"), CoffeeOrderContext(2, 1)),
        (factory.get_flavor("卡布奇诺"), CoffeeOrderContext(5, 1)),  # 复用
        (factory.get_flavor("美式"), CoffeeOrderContext(3, 3)),
        (factory.get_flavor("拿铁"), CoffeeOrderContext(8, 2)),      # 复用
        (factory.get_flavor("卡布奇诺"), CoffeeOrderContext(12, 1)), # 复用
    ]

    print("\n上菜:")
    for flavor, context in orders:
        for _ in range(context.quantity):
            flavor.serve(context.table_number)

    print(f"\n订单数: {sum(c.quantity for _, c in orders)}")
    print(f"享元对象数: {factory.total_flavors()}")

    print("\n=== 享元模式：森林渲染 ===\n")
    forest = Forest()
    # 种 100 棵树，但只有 3 种类型
    import random
    tree_specs = [
        ("橡树", "绿色", "粗糙"),
        ("松树", "深绿", "针叶"),
        ("桦树", "白色", "光滑"),
    ]
    for _ in range(100):
        name, color, texture = random.choice(tree_specs)
        forest.plant_tree(random.randint(0, 1000), random.randint(0, 1000),
                          name, color, texture)
    forest.draw()
    print(f"\n树木总数: {len(forest._trees)}")
    print(f"树木类型数: {len(TreeFactory._types)}")

适用场景

  • 系统中有大量相同或相似的对象，内存成为瓶颈
  • 对象的大部分状态可以变为外在状态（参数化）
  • 字符串常量池、数据库连接池、线程池
  • 游戏中的大量相似实体（树木、粒子、NPC）
  • 文档编辑器中的字符渲染

优点

  • 大幅减少内存消耗：相同对象只创建一次
  • 提高性能：减少对象创建和 GC 开销
  • 内外分离：清晰区分共享与非共享状态

缺点

  • 设计复杂：需要仔细区分内在状态和外在状态
  • 线程安全：共享对象需要考虑并发访问
  • 状态管理：外在状态的计算和传递可能增加代码复杂度

与其他模式的关系

  • 享元 vs 单例：单例限制全局一个实例，享元管理一组可共享实例
  • 享元 vs 对象池：对象池管理可复用对象的生命周期，享元关注状态共享
  • 享元 vs 组合：享元常与组合模式配合使用（如文档树的字符节点）

实际应用

  • Java String 常量池：字符串字面量通过享元共享
  • Python intern()：sys.intern() 将字符串放入享元池
  • Java Integer 缓存：-128 到 127 的 Integer 对象被缓存共享
  • 游戏引擎：Unity/Unreal 中大量使用享元渲染相同类型的对象
  • CSS 类：多个 DOM 元素共享同一 CSS 类定义
  • 数据库连接池：连接对象复用，避免频繁创建销毁
