迭代器模式

约 1649 字大约 6 分钟

2025-09-24

定义

迭代器模式提供一种方法顺序访问一个聚合对象中的各个元素，而又不需要暴露该对象的内部表示。迭代器模式将遍历的职责从聚合对象中分离出来，使聚合对象专注于数据存储，迭代器专注于遍历。

结构

  • Iterator（迭代器接口）：定义访问和遍历元素的接口，通常包含 hasNext() 和 next() 方法。
  • ConcreteIterator（具体迭代器）：实现迭代器接口，跟踪当前遍历位置。
  • Aggregate（聚合接口）：定义创建迭代器的方法（createIterator() / iterator()）。
  • ConcreteAggregate（具体聚合）：实现聚合接口，返回一个对应于自身的迭代器。

现代编程语言大多内置了迭代器支持（如 Java 的 Iterable、Python 的 __iter__ / __next__、C++ 的 STL 迭代器）。

UML 类图（文本描述）

Iterator <|-- ConcreteIterator
    |-- hasNext(): bool
    |-- next(): Element

Aggregate <|-- ConcreteAggregate
    |-- createIterator(): Iterator

ConcreteIterator --> ConcreteAggregate（关联：持有聚合引用，访问其内部元素）
ConcreteAggregate --> Iterator（创建）

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <vector>
#include <memory>
#include <stdexcept>

// ========== 聚合接口 ==========
template<typename T>
class IIterator {
public:
    virtual ~IIterator() = default;
    virtual bool hasNext() const = 0;
    virtual T next() = 0;
};

template<typename T>
class ICollection {
public:
    virtual ~ICollection() = default;
    virtual std::unique_ptr<IIterator<T>> createIterator() = 0;
    virtual int size() const = 0;
};

// ========== 具体聚合：餐厅菜单 ==========
class MenuItem {
private:
    std::string name_;
    std::string description_;
    double price_;
public:
    MenuItem(std::string name, std::string desc, double price)
        : name_(std::move(name)), description_(std::move(desc)), price_(price) {}

    const std::string& getName() const { return name_; }
    const std::string& getDescription() const { return description_; }
    double getPrice() const { return price_; }
};

class Menu : public ICollection<MenuItem> {
private:
    std::vector<MenuItem> items_;
    std::string name_;

    // ========== 具体迭代器 ==========
    class MenuIterator : public IIterator<MenuItem> {
    private:
        const std::vector<MenuItem>& items_;
        int position_ = 0;
    public:
        explicit MenuIterator(const std::vector<MenuItem>& items)
            : items_(items) {}

        bool hasNext() const override {
            return position_ < static_cast<int>(items_.size());
        }
        MenuItem next() override {
            if (!hasNext()) throw std::out_of_range("No more items");
            return items_[position_++];
        }
    };

public:
    explicit Menu(std::string name) : name_(std::move(name)) {}

    void addItem(const std::string& name, const std::string& desc, double price) {
        items_.emplace_back(name, desc, price);
    }

    std::unique_ptr<IIterator<MenuItem>> createIterator() override {
        return std::make_unique<MenuIterator>(items_);
    }

    int size() const override { return static_cast<int>(items_.size()); }
    const std::string& getName() const { return name_; }
};

// ========== 具体聚合： pancakes 早餐菜单（使用数组） ==========
class PancakeMenu : public ICollection<MenuItem> {
private:
    static constexpr int MAX_ITEMS = 6;
    MenuItem* items_[MAX_ITEMS];
    int position_ = 0;

    class PancakeIterator : public IIterator<MenuItem> {
    private:
        MenuItem* items_[MAX_ITEMS];
        int position_ = 0;
        int size_;
    public:
        PancakeIterator(MenuItem* items[], int size) : size_(size) {
            for (int i = 0; i < size; ++i) items_[i] = items[i];
        }
        bool hasNext() const override { return position_ < size_; }
        MenuItem next() override {
            if (!hasNext()) throw std::out_of_range("No more items");
            return *items_[position_++];
        }
    };

public:
    PancakeMenu() {
        for (int i = 0; i < MAX_ITEMS; ++i) items_[i] = nullptr;
    }
    ~PancakeMenu() {
        for (int i = 0; i < position_; ++i) delete items_[i];
    }

    void addItem(const std::string& name, const std::string& desc, double price) {
        if (position_ < MAX_ITEMS)
            items_[position_++] = new MenuItem(name, desc, price);
    }

    std::unique_ptr<IIterator<MenuItem>> createIterator() override {
        return std::make_unique<PancakeIterator>(items_, position_);
    }

    int size() const override { return position_; }
};

void printMenu(ICollection<MenuItem>* menu) {
    std::cout << "\n=== " << menu->getName() << " (共 "
              << menu->size() << " 项) ===\n";
    auto it = menu->createIterator();
    while (it->hasNext()) {
        MenuItem item = it->next();
        std::cout << "  " << item.getName() << " - ¥" << item.getPrice()
                  << ": " << item.getDescription() << "\n";
    }
}

int main() {
    Menu dinner("晚餐菜单");
    dinner.addItem("意大利面", "经典番茄肉酱面", 38.0);
    dinner.addItem("凯撒沙拉", "新鲜罗马生菜配面包丁", 28.0);
    dinner.addItem("牛排", "澳洲谷饲 200g", 128.0);

    PancakeMenu breakfast;
    breakfast.addItem("松饼套餐", "3块松饼配枫糖浆", 22.0);
    breakfast.addItem("煎蛋三明治", "培根煎蛋配吐司", 25.0);

    printMenu(&dinner);
    printMenu(&breakfast);

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from typing import Any, Iterator, List


# ========== 经典迭代器实现 ==========
class Iterator(ABC):
    @abstractmethod
    def has_next(self) -> bool: ...

    @abstractmethod
    def next(self) -> Any: ...

    @abstractmethod
    def reset(self) -> None: ...


class Collection(ABC):
    @abstractmethod
    def create_iterator(self) -> Iterator: ...


class WordsCollection(Collection):
    def __init__(self, words: List[str]) -> None:
        self._words = words

    def create_iterator(self) -> Iterator:
        return AlphabeticalIterator(self)

    @property
    def words(self) -> List[str]:
        return self._words

    def __len__(self) -> int:
        return len(self._words)


class AlphabeticalIterator(Iterator):
    def __init__(self, collection: WordsCollection, reverse: bool = False) -> None:
        self._collection = collection
        self._reverse = reverse
        self._position = -1 if reverse else 0

    def has_next(self) -> bool:
        if self._reverse:
            return self._position >= -len(self._collection)
        return self._position < len(self._collection)

    def next(self) -> str:
        if not self.has_next():
            raise StopIteration
        word = self._collection.words[self._position]
        self._position += -1 if self._reverse else 1
        return word

    def reset(self) -> None:
        self._position = -1 if self._reverse else 0


# ========== 实际应用：二叉树遍历 ==========
class TreeNode:
    def __init__(self, value: Any) -> None:
        self.value = value
        self.left: "TreeNode | None" = None
        self.right: "TreeNode | None" = None


class InOrderIterator:
    """中序遍历迭代器"""

    def __init__(self, root: TreeNode | None) -> None:
        self._stack: list[TreeNode] = []
        self._current = root
        # 初始化：将左路径压栈
        self._push_left(root)

    def _push_left(self, node: TreeNode | None) -> None:
        while node:
            self._stack.append(node)
            node = node.left

    def has_next(self) -> bool:
        return len(self._stack) > 0

    def next(self) -> Any:
        if not self.has_next():
            raise StopIteration
        node = self._stack.pop()
        self._push_left(node.right)
        return node.value


class PreOrderIterator:
    """先序遍历迭代器"""

    def __init__(self, root: TreeNode | None) -> None:
        self._stack: list[TreeNode] = [root] if root else []

    def has_next(self) -> bool:
        return len(self._stack) > 0

    def next(self) -> Any:
        if not self.has_next():
            raise StopIteration
        node = self._stack.pop()
        if node.right:
            self._stack.append(node.right)
        if node.left:
            self._stack.append(node.left)
        return node.value


# ========== Pythonic 方式：生成器 ==========
def in_order_generator(node: TreeNode | None):
    """中序遍历生成器（Python 最优雅的实现）"""
    if node is None:
        return
    yield from in_order_generator(node.left)
    yield node.value
    yield from in_order_generator(node.right)


if __name__ == "__main__":
    print("=== 迭代器模式：单词集合 ===\n")
    collection = WordsCollection(["banana", "apple", "cherry", "date"])

    print("正向遍历:")
    it = collection.create_iterator()
    while it.has_next():
        print(f"  {it.next()}")

    print("\n反向遍历:")
    reverse_it = AlphabeticalIterator(collection, reverse=True)
    while reverse_it.has_next():
        print(f"  {reverse_it.next()}")

    print("\nPython 原生迭代器协议:")
    for word in collection.words:
        print(f"  {word}")

    print("\n=== 迭代器模式：二叉树遍历 ===\n")
    # 构建二叉搜索树:
    #        4
    #       / \
    #      2   6
    #     / \ / \
    #    1  3 5  7
    root = TreeNode(4)
    root.left = TreeNode(2)
    root.right = TreeNode(6)
    root.left.left = TreeNode(1)
    root.left.right = TreeNode(3)
    root.right.left = TreeNode(5)
    root.right.right = TreeNode(7)

    print("中序遍历（迭代器）:")
    it = InOrderIterator(root)
    while it.has_next():
        print(f"  {it.next()}", end=" ")
    print()

    print("\n先序遍历（迭代器）:")
    it = PreOrderIterator(root)
    while it.has_next():
        print(f"  {it.next()}", end=" ")
    print()

    print("\n中序遍历（生成器）:")
    print("  ", list(in_order_generator(root)))

适用场景

  • 需要访问聚合对象的内容但不暴露其内部结构
  • 需要为聚合对象提供多种遍历方式
  • 需要统一遍历不同结构的聚合对象
  • 延迟加载：迭代器可以在需要时才获取下一个元素

优点

  • 单一职责：将遍历逻辑从聚合对象中分离
  • 开闭原则：新增聚合类型或遍历方式不影响现有代码
  • 封装内部结构：客户端无需知道聚合如何存储数据
  • 统一接口：不同聚合可以使用相同的遍历接口

缺点

  • 对于简单集合可能过度：简单的 for-each 可能更直观
  • 并发问题：遍历过程中聚合被修改可能导致不一致
  • 额外开销：迭代器对象本身需要内存和创建成本

与其他模式的关系

  • 迭代器 vs 组合：迭代器常用于遍历组合树结构
  • 迭代器 vs 访问者：迭代器提供遍历能力，访问者在遍历时执行操作
  • 迭代器 vs 工厂：聚合使用工厂方法创建迭代器

实际应用

  • STL 迭代器：C++ 标准库的核心抽象
  • Java Iterator / Iterable：所有 Collection 的基础
  • Python __iter__ / __next__：迭代器协议
  • 生成器：Python 的 yield 是最简洁的迭代器实现
  • 数据库游标：Cursor 模式是迭代器的应用
  • 文件读取：逐行读取文件就是迭代器
