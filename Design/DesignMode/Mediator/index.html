中介者模式

约 1802 字大约 6 分钟

2025-09-24

定义

中介者模式用一个中介对象来封装一系列对象之间的交互。中介者使各对象不需要显式地相互引用，从而使其耦合松散，而且可以独立地改变它们之间的交互。核心思想是"将网状关系转为星形关系"。

结构

  • Mediator（中介者接口）：定义同事对象与中介者之间的通信接口。
  • ConcreteMediator（具体中介者）：实现中介者接口，协调各同事对象之间的交互。它知道所有同事对象。
  • Colleague（同事接口）：定义同事类的公共接口，每个同事都知道中介者对象。
  • ConcreteColleague（具体同事）：每个同事在需要与其他同事通信时，不直接通信，而是通知中介者，由中介者协调。

中介者模式与观察者模式的区别：观察者模式是一对多广播，中介者模式是多对多协调。中介者模式中所有交互都经过中心节点。

UML 类图（文本描述）

Mediator <|-- ConcreteMediator
    |-- notify(sender, event)

Colleague --> Mediator（关联：每个同事持有中介者引用）
Colleague <|-- ConcreteColleagueA
Colleague <|-- ConcreteColleagueB
Colleague <|-- ConcreteColleagueC

ConcreteMediator --> ConcreteColleagueA（关联：知道所有同事）
ConcreteMediator --> ConcreteColleagueB
ConcreteMediator --> ConcreteColleagueC

交互方式（不用中介者 vs 用中介者）:
  不用: A <--> B, A <--> C, B <--> C  (网状，N*(N-1)/2 条连接)
  用:   A --> Mediator --> B/C         (星形，N 条连接)

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <unordered_map>
#include <vector>

// ========== 前向声明 ==========
class Mediator;
class Component;

// ========== 中介者接口 ==========
class Mediator {
public:
    virtual ~Mediator() = default;
    virtual void notify(Component* sender, const std::string& event) = 0;
};

// ========== 同事基类 ==========
class Component {
protected:
    Mediator* mediator_;
public:
    explicit Component(Mediator* mediator = nullptr) : mediator_(mediator) {}
    virtual ~Component() = default;
    void setMediator(Mediator* mediator) { mediator_ = mediator; }
};

// ========== 具体同事：按钮 ==========
class Button : public Component {
private:
    std::string name_;
public:
    Button(const std::string& name, Mediator* m = nullptr)
        : Component(m), name_(name) {}

    void click() {
        std::cout << "[Button] " << name_ << " 被点击\n";
        mediator_->notify(this, "click");
    }
};

// ========== 具体同事：文本框 ==========
class TextBox : public Component {
private:
    std::string name_;
    std::string content_;
public:
    TextBox(const std::string& name, Mediator* m = nullptr)
        : Component(m), name_(name) {}

    void setText(const std::string& text) {
        content_ = text;
        std::cout << "[TextBox] " << name_ << " 内容: " << text << "\n";
        mediator_->notify(this, "textChanged");
    }
    const std::string& getContent() const { return content_; }
};

// ========== 具体同事：复选框 ==========
class Checkbox : public Component {
private:
    std::string name_;
    bool checked_ = false;
public:
    Checkbox(const std::string& name, Mediator* m = nullptr)
        : Component(m), name_(name) {}

    void toggle() {
        checked_ = !checked_;
        std::cout << "[Checkbox] " << name_
                  << (checked_ ? " 已勾选" : " 已取消") << "\n";
        mediator_->notify(this, checked_ ? "check" : "uncheck");
    }
    bool isChecked() const { return checked_; }
};

// ========== 具体同事：标签 ==========
class Label : public Component {
private:
    std::string name_;
    std::string text_;
public:
    Label(const std::string& name, Mediator* m = nullptr)
        : Component(m), name_(name) {}

    void setText(const std::string& text) {
        text_ = text;
        std::cout << "[Label] " << name_ << ": " << text << "\n";
    }
    void clear() {
        text_.clear();
        std::cout << "[Label] " << name_ << ": (空)\n";
    }
};

// ========== 具体中介者：登录对话框 ==========
class AuthDialog : public Mediator {
private:
    Button* loginButton_;
    Button* guestButton_;
    TextBox* usernameBox_;
    TextBox* passwordBox_;
    Checkbox* rememberMe_;
    Label* errorLabel_;

public:
    AuthDialog() {
        loginButton_ = new Button("登录", this);
        guestButton_ = new Button("访客登录", this);
        usernameBox_ = new TextBox("用户名", this);
        passwordBox_ = new TextBox("密码", this);
        rememberMe_ = new Checkbox("记住我", this);
        errorLabel_ = new Label("错误提示", this);
        errorLabel_->clear();
    }
    ~AuthDialog() {
        delete loginButton_;
        delete guestButton_;
        delete usernameBox_;
        delete passwordBox_;
        delete rememberMe_;
        delete errorLabel_;
    }

    void notify(Component* sender, const std::string& event) override {
        if (sender == guestButton_) {
            usernameBox_->setText("Guest");
            passwordBox_->setText("");
            loginButton_->click();
        }
        if (sender == loginButton_) {
            if (usernameBox_->getContent().empty() ||
                passwordBox_->getContent().empty()) {
                errorLabel_->setText("请填写用户名和密码！");
            } else if (!rememberMe_->isChecked()) {
                errorLabel_->setText("登录成功（未记住密码）");
            } else {
                errorLabel_->setText("登录成功（已保存密码）");
            }
        }
        if (sender == usernameBox_ || sender == passwordBox_) {
            if (event == "textChanged" &&
                !usernameBox_->getContent().empty() &&
                !passwordBox_->getContent().empty()) {
                errorLabel_->clear();
            }
        }
    }
};

int main() {
    std::cout << "=== 中介者模式：登录对话框 ===\n\n";
    AuthDialog dialog;

    std::cout << "--- 场景 1：直接登录（未填写） ---\n";
    // 模拟点击登录按钮

    std::cout << "\n--- 场景 2：填写信息后登录 ---\n";
    // 模拟填写用户名

    std::cout << "\n--- 场景 3：访客登录 ---\n";
    // 模拟点击访客登录

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from typing import Dict, List, Any


# ========== 中介者接口 ==========
class Mediator(ABC):
    @abstractmethod
    def notify(self, sender: "Colleague", event: str) -> None: ...


# ========== 同事基类 ==========
class Colleague:
    def __init__(self, mediator: Mediator | None = None) -> None:
        self._mediator = mediator

    @property
    def mediator(self) -> Mediator | None:
        return self._mediator

    @mediator.setter
    def mediator(self, value: Mediator) -> None:
        self._mediator = value


# ========== 实际应用：聊天室 ==========
class ChatMediator(Mediator):
    def __init__(self) -> None:
        self._users: Dict[str, "User"] = {}

    def add_user(self, user: "User") -> None:
        self._users[user.name] = user
        user.mediator = self

    def notify(self, sender: Colleague, event: str) -> None:
        if isinstance(sender, User):
            if event == "sendMessage":
                self._broadcast(sender)
            elif event == "addUser":
                self._announce_join(sender)
            elif event == "privateMessage":
                pass  # 私信单独处理

    def _broadcast(self, sender: "User") -> None:
        for name, user in self._users.items():
            if name != sender.name:
                user.receive(sender.name, sender.last_message)

    def _announce_join(self, user: "User") -> None:
        for name, u in self._users.items():
            if name != user.name:
                u.receive("系统", f"{user.name} 加入了聊天室")

    def send_private(self, sender: "User", target_name: str, message: str) -> None:
        target = self._users.get(target_name)
        if target:
            target.receive(f"{sender.name}(私信)", message)
        else:
            sender.receive("系统", f"用户 {target_name} 不存在")


class User(Colleague):
    def __init__(self, name: str) -> None:
        super().__init__()
        self._name = name
        self._last_message = ""

    @property
    def name(self) -> str:
        return self._name

    @property
    def last_message(self) -> str:
        return self._last_message

    def send_message(self, message: str) -> None:
        self._last_message = message
        print(f"[{self._name}] 发送: {message}")
        self.mediator.notify(self, "sendMessage")

    def send_private(self, target: str, message: str) -> None:
        if self.mediator:
            assert isinstance(self.mediator, ChatMediator)
            self.mediator.send_private(self, target, message)

    def receive(self, sender: str, message: str) -> None:
        print(f"  → [{self._name}] 收到来自 {sender}: {message}")


# ========== 实际应用：航班调度 ==========
class Flight:
    def __init__(self, flight_id: str) -> None:
        self._id = flight_id
        self._status = "等待"

    @property
    def flight_id(self) -> str:
        return self._id

    @property
    def status(self) -> str:
        return self._status

    @status.setter
    def status(self, value: str) -> None:
        self._status = value


class ControlTower(Mediator):
    """塔台：航班调度的中介者"""

    def __init__(self) -> None:
        self._runways: Dict[str, bool] = {"RWY-01": True, "RWY-02": True}
        self._flights: Dict[str, Flight] = {}

    def register(self, flight: Flight) -> None:
        self._flights[flight.flight_id] = flight

    def request_landing(self, flight: Flight) -> None:
        for rwy, available in self._runways.items():
            if available:
                self._runways[rwy] = False
                print(f"  ✓ {flight.flight_id} 分配跑道 {rwy}，允许降落")
                flight.status = "降落中"
                return
        print(f"  ✗ {flight.flight_id} 无可用跑道，请盘旋等待")
        flight.status = "等待"

    def notify(self, sender: Colleague, event: str) -> None: ...


if __name__ == "__main__":
    print("=== 中介者模式：聊天室 ===\n")
    mediator = ChatMediator()

    alice = User("Alice")
    bob = User("Bob")
    charlie = User("Charlie")

    mediator.add_user(alice)
    mediator.add_user(bob)
    mediator.add_user(charlie)

    alice.send_message("大家好！")
    bob.send_message("嗨 Alice！")
    charlie.send_private("Bob", "晚上一起吃饭吗？")

    print("\n=== 中介者模式：航班调度 ===\n")
    tower = ControlTower()
    flights = [Flight(f"CA{i}") for i in range(101, 105)]
    for f in flights:
        tower.register(f)
        tower.request_landing(f)

适用场景

  • 多个类之间存在复杂的网状依赖关系
  • 希望减少类之间的直接通信，降低耦合
  • 组件的交互行为集中管理，便于维护和扩展
  • GUI 组件之间的协调（对话框中各控件的联动）

优点

  • 解耦同事对象：同事之间不直接通信
  • 集中控制：交互逻辑集中在中介者中，便于理解和修改
  • 单一职责：每个同事只关注自身功能
  • 开闭原则：新增同事类无需修改已有同事

缺点

  • 中介者可能变得庞大：随着同事增多，中介者可能过于复杂
  • 过度集中：中介者成为系统的瓶颈和单点
  • 调试困难：所有交互都经过中介者，追踪调用链较难

与其他模式的关系

  • 中介者 vs 观察者：观察者是一对多广播，中介者是多对多协调
  • 中介者 vs 外观：外观单向简化访问，中介者双向协调
  • 中介者 vs 命令：中介者可以与命令模式配合，用命令封装交互

实际应用

  • GUI 框架：对话框中各控件的联动由 Dialog 类协调
  • MVC 模式：Controller 就是 Model 和 View 之间的中介者
  • 消息中间件：RabbitMQ/Kafka 是生产者与消费者的中介
  • 航班调度：塔台协调所有航班的起降
  • 聊天室：服务器作为用户之间的中介
  • Event Bus：事件总线是模块间解耦的中介
