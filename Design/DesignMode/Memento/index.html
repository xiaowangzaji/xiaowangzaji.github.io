备忘录模式

约 1649 字大约 6 分钟

2025-09-24

定义

备忘录模式在不破坏封装性的前提下，捕获一个对象的内部状态，并在该对象之外保存这个状态。这样以后就可将该对象恢复到原先保存的状态。核心思想是"在不暴露内部状态的情况下保存和恢复对象状态"。

结构

  • Originator（发起人）：需要保存状态的对象。负责创建备忘录（保存当前状态）和使用备忘录（恢复状态）。
  • Memento（备忘录）：存储 Originator 的内部状态。备忘录应当对除 Originator 之外的对象不可见，以防止破坏封装性。
  • Caretaker（管理者）：负责保存备忘录，但不查看或修改备忘录的内容。它可以存储多个备忘录形成历史记录。

备忘录模式的关键在于封装性：只有 Originator 可以访问备忘录的内部状态，Caretaker 只能传递备忘录对象。

UML 类图（文本描述）

Originator
    |-- save() -> Memento         // 创建备忘录
    |-- restore(Memento)          // 恢复状态
    |-- 其他业务方法...

Memento
    |-- 内部状态（仅 Originator 可访问）

Caretaker --> Memento（持有备忘录引用，但不访问其内部）

使用流程:
  1. Originator 创建 Memento 保存状态
  2. 将 Memento 交给 Caretaker 保管
  3. Originator 状态发生变化
  4. 需要恢复时，从 Caretaker 取回 Memento
  5. Originator 用 Memento 恢复状态

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <vector>
#include <memory>
#include <stack>

// ========== 备忘录 ==========
class Memento {
private:
    std::string text_;
    int cursorPos_;
    std::string clipboard_;

    // 只允许 Originator 访问
    friend class TextEditor;

public:
    Memento(std::string text, int cursor, std::string clip)
        : text_(std::move(text)), cursorPos_(cursor),
          clipboard_(std::move(clip)) {}
};

// ========== 发起人：文本编辑器 ==========
class TextEditor {
private:
    std::string text_;
    int cursorPos_;
    std::string clipboard_;

public:
    TextEditor() : cursorPos_(0) {}

    void type(const std::string& str) {
        text_.insert(cursorPos_, str);
        cursorPos_ += static_cast<int>(str.size());
    }

    void backspace(int count = 1) {
        if (cursorPos_ >= count) {
            text_.erase(cursorPos_ - count, count);
            cursorPos_ -= count;
        }
    }

    void copy(int start, int end) {
        if (start >= 0 && end <= static_cast<int>(text_.size())) {
            clipboard_ = text_.substr(start, end - start);
        }
    }

    void paste() {
        text_.insert(cursorPos_, clipboard_);
        cursorPos_ += static_cast<int>(clipboard_.size());
    }

    void display() const {
        std::cout << "  [" << text_ << "] (光标: " << cursorPos_
                  << ") [剪贴板: " << clipboard_ << "]\n";
    }

    // 创建备忘录
    Memento save() const {
        return Memento(text_, cursorPos_, clipboard_);
    }

    // 从备忘录恢复
    void restore(const Memento& memento) {
        text_ = memento.text_;
        cursorPos_ = memento.cursorPos_;
        clipboard_ = memento.clipboard_;
    }
};

// ========== 管理者：撤销历史 ==========
class History {
private:
    std::stack<Memento> states_;
    static constexpr int MAX_STATES = 10;
public:
    void push(Memento state) {
        if (static_cast<int>(states_.size()) >= MAX_STATES) {
            // 简化：直接弹出底部（实际可用双端队列）
            std::stack<Memento> temp;
            states_.pop();  // 丢弃最旧
            states_.swap(temp);
        }
        states_.push(std::move(state));
        std::cout << "  [保存状态] 历史栈深度: " << states_.size() << "\n";
    }

    Memento pop() {
        if (states_.empty()) {
            throw std::runtime_error("无历史可撤销");
        }
        auto m = std::move(states_.top());
        states_.pop();
        std::cout << "  [撤销] 历史栈深度: " << states_.size() << "\n";
        return m;
    }

    bool empty() const { return states_.empty(); }
};

int main() {
    TextEditor editor;
    History history;

    std::cout << "=== 备忘录模式：文本编辑器 ===\n";

    // 初始状态
    history.push(editor.save());
    editor.display();

    // 输入内容
    std::cout << "\n--- 输入 Hello ---\n";
    editor.type("Hello");
    editor.display();
    history.push(editor.save());

    // 继续输入
    std::cout << "\n--- 输入 World ---\n";
    editor.type(" World");
    editor.display();
    history.push(editor.save());

    // 复制粘贴
    std::cout << "\n--- 复制粘贴 ---\n";
    editor.copy(0, 5);
    editor.paste();
    editor.display();
    history.push(editor.save());

    // 撤销
    std::cout << "\n--- 撤销 ---\n";
    editor.restore(history.pop());
    editor.display();

    std::cout << "\n--- 再次撤销 ---\n";
    editor.restore(history.pop());
    editor.display();

    std::cout << "\n--- 第三次撤销 ---\n";
    editor.restore(history.pop());
    editor.display();

    return 0;
}

代码实现（Python 示例）

from __future__ import annotations
from dataclasses import dataclass, field
from typing import List, Any
import copy


# ========== 备忘录 ==========
@dataclass
class Memento:
    """存储发起人状态快照"""
    state: dict

    @property
    def name(self) -> str:
        return self.state.get("name", "")

    @property
    def level(self) -> int:
        return self.state.get("level", 0)

    @property
    def hp(self) -> int:
        return self.state.get("hp", 0)

    @property
    def inventory(self) -> list:
        return self.state.get("inventory", [])


# ========== 发起人：游戏角色 ==========
class GameCharacter:
    def __init__(self, name: str, level: int = 1) -> None:
        self._name = name
        self._level = level
        self._hp = 100
        self._max_hp = 100
        self._inventory: list[str] = ["木剑"]

    @property
    def name(self) -> str:
        return self._name

    def display(self) -> None:
        print(f"  {self._name} Lv.{self._level} HP:{self._hp}/{self._max_hp} "
              f"背包: {self._inventory}")

    def take_damage(self, amount: int) -> None:
        self._hp = max(0, self._hp - amount)
        print(f"  💥 {self._name} 受到 {amount} 点伤害，剩余 HP: {self._hp}")

    def heal(self, amount: int) -> None:
        self._hp = min(self._max_hp, self._hp + amount)
        print(f"  💚 {self._name} 恢复 {amount} 点 HP，当前 HP: {self._hp}")

    def level_up(self) -> None:
        self._level += 1
        self._max_hp += 20
        self._hp = self._max_hp
        print(f"  🎉 {self._name} 升到 Lv.{self._level}！HP 上限提升")

    def pick_up(self, item: str) -> None:
        self._inventory.append(item)
        print(f"  🎒 {self._name} 拾取: {item}")

    def use_item(self, item: str) -> bool:
        if item in self._inventory:
            self._inventory.remove(item)
            print(f"  🗑 {self._name} 使用: {item}")
            return True
        return False

    # 创建备忘录
    def save(self) -> Memento:
        return Memento({
            "name": self._name,
            "level": self._level,
            "hp": self._hp,
            "max_hp": self._max_hp,
            "inventory": list(self._inventory),
        })

    # 从备忘录恢复
    def restore(self, memento: Memento) -> None:
        self._name = memento.state["name"]
        self._level = memento.state["level"]
        self._hp = memento.state["hp"]
        self._max_hp = memento.state["max_hp"]
        self._inventory = list(memento.state["inventory"])


# ========== 管理者：存档系统 ==========
class SaveSystem:
    def __init__(self, max_saves: int = 5) -> None:
        self._saves: List[Memento] = []
        self._max_saves = max_saves

    def save(self, memento: Memento) -> None:
        if len(self._saves) >= self._max_saves:
            self._saves.pop(0)  # 删除最早的存档
        self._saves.append(memento)
        print(f"  💾 已存档 (共 {len(self._saves)} 个存档)")

    def load(self) -> Memento:
        if not self._saves:
            raise RuntimeError("没有存档")
        memento = self._saves.pop()
        print(f"  📂 加载存档 (剩余 {len(self._saves)} 个存档)")
        return memento

    @property
    def save_count(self) -> int:
        return len(self._saves)


if __name__ == "__main__":
    print("=== 备忘录模式：游戏存档系统 ===\n")

    hero = GameCharacter("勇者", level=1)
    save_system = SaveSystem(max_saves=3)

    # 初始状态存档
    print("--- 创建角色 ---")
    hero.display()
    save_system.save(hero.save())

    # 冒险过程
    print("\n--- 打怪升级 ---")
    hero.take_damage(30)
    hero.pick_up("铁剑")
    save_system.save(hero.save())

    print("\n--- 继续冒险 ---")
    hero.level_up()
    hero.pick_up("盾牌")
    hero.take_damage(50)
    save_system.save(hero.save())

    print("\n--- Boss 战 ---")
    hero.take_damage(80)
    hero.display()

    # 状态很差，读档重来
    print("\n--- 读档重来 ---")
    hero.restore(save_system.load())
    hero.display()

    print("\n--- 再次挑战 ---")
    hero.heal(50)
    hero.use_item("铁剑")  # 装备替换
    hero.display()

适用场景

  • 需要保存和恢复对象状态（撤销/重做、存档/读档）
  • 直接暴露对象内部状态会破坏封装性
  • 需要保存对象状态的快照，但保存和恢复逻辑不应暴露给外部
  • 游戏存档、编辑器撤销、事务回滚

优点

  • 保持封装：Originator 以外的对象无法访问备忘录内部状态
  • 简化 Originator：状态管理职责转移给了 Caretaker
  • 历史管理：Caretaker 可以实现复杂的历史管理策略

缺点

  • 内存消耗：每次保存都复制整个状态，频繁保存可能占用大量内存
  • 动态语言难以完全封装：Python/JS 等语言无法严格限制对备忘录的访问
  • 生命周期管理：如果 Originator 发生变化，旧备忘录可能不兼容

与其他模式的关系

  • 备忘录 vs 命令：命令通过反向操作撤销，备忘录通过恢复状态撤销
  • 备忘录 vs 原型：原型用于创建对象的副本，备忘录用于保存和恢复状态
  • 备忘录 vs 迭代器：迭代器遍历集合，备忘录保存状态快照

实际应用

  • 编辑器撤销：文本/图像编辑器的 Ctrl+Z
  • 游戏存档：RPG 游戏的存档/读档系统
  • 数据库事务：事务日志 + 回滚
  • Git：每次 commit 就是一次备忘录
  • 虚拟机快照：VMware/VirtualBox 的快照功能
  • 浏览器历史：前进/后退按钮
