观察者模式

约 1690 字大约 6 分钟

2025-09-24

定义

观察者模式定义了一种一对多的依赖关系，让多个观察者对象同时监听某一个主题对象。当主题对象的状态发生变化时，会通知所有观察者，使它们能够自动更新自己。这是事件驱动编程和响应式编程的基础模式。

结构

  • Subject（主题/被观察者）：维护观察者列表，提供注册、移除、通知观察者的方法。
  • Observer（观察者）：定义更新接口，当主题状态改变时接收通知。
  • ConcreteSubject（具体主题）：存储具体状态，状态变化时向所有注册的观察者发送通知。
  • ConcreteObserver（具体观察者）：实现 Observer 接口，保持一个指向 ConcreteSubject 的引用，在收到通知时更新自身状态。

观察者模式支持推模型（主题主动推送数据给观察者）和拉模型（主题通知观察者，观察者自行拉取数据）。

UML 类图（文本描述）

Subject --> Observer（聚合：持有观察者列表）
Subject <|-- ConcreteSubject
Observer <|-- ConcreteObserverA
Observer <|-- ConcreteObserverB

ConcreteSubject 状态改变时:
  for observer in observers:
      observer.update()

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <memory>

// ========== 观察者接口 ==========
class IObserver {
public:
    virtual ~IObserver() = default;
    virtual void update(float temperature, float humidity, float pressure) = 0;
};

// ========== 主题接口 ==========
class ISubject {
public:
    virtual ~ISubject() = default;
    virtual void attach(IObserver* observer) = 0;
    virtual void detach(IObserver* observer) = 0;
    virtual void notify() = 0;
};

// ========== 具体主题：气象站 ==========
class WeatherStation : public ISubject {
private:
    std::vector<IObserver*> observers_;
    float temperature_;
    float humidity_;
    float pressure_;

public:
    void attach(IObserver* observer) override {
        observers_.push_back(observer);
    }
    void detach(IObserver* observer) override {
        observers_.erase(
            std::remove(observers_.begin(), observers_.end(), observer),
            observers_.end()
        );
    }
    void notify() override {
        for (auto* obs : observers_) {
            obs->update(temperature_, humidity_, pressure_);
        }
    }
    void setMeasurements(float temp, float hum, float press) {
        temperature_ = temp;
        humidity_ = hum;
        pressure_ = press;
        std::cout << "[气象站] 数据更新 -> 温度=" << temp
                  << "°C 湿度=" << hum << "% 气压=" << press << "hPa\n";
        notify();
    }
};

// ========== 具体观察者：当前状况显示 ==========
class CurrentConditionsDisplay : public IObserver {
private:
    float temperature_;
    float humidity_;
public:
    void update(float temp, float hum, float press) override {
        temperature_ = temp;
        humidity_ = hum;
        std::cout << "  [当前状况] 温度: " << temperature_
                  << "°C | 湿度: " << humidity_ << "%\n";
    }
};

// ========== 具体观察者：统计显示 ==========
class StatisticsDisplay : public IObserver {
private:
    float maxTemp_ = -999, minTemp_ = 999, sumTemp_ = 0;
    int count_ = 0;
public:
    void update(float temp, float hum, float press) override {
        sumTemp_ += temp;
        count_++;
        if (temp > maxTemp_) maxTemp_ = temp;
        if (temp < minTemp_) minTemp_ = temp;
        std::cout << "  [统计] 平均=" << (sumTemp_ / count_)
                  << "°C 最高=" << maxTemp_
                  << "°C 最低=" << minTemp_ << "°C\n";
    }
};

// ========== 具体观察者：预警显示 ==========
class WarningDisplay : public IObserver {
public:
    void update(float temp, float hum, float press) override {
        if (temp > 40) std::cout << "  ⚠ [高温预警] 温度超过 40°C！\n";
        if (press < 980) std::cout << "  ⚠ [低压预警] 气压低于 980hPa，可能有风暴！\n";
        if (hum > 90) std::cout << "  ⚠ [高湿预警] 湿度超过 90%，注意防潮！\n";
    }
};

int main() {
    WeatherStation station;
    CurrentConditionsDisplay current;
    StatisticsDisplay stats;
    WarningDisplay warning;

    station.attach(&current);
    station.attach(&stats);
    station.attach(&warning);

    std::cout << "=== 气象数据更新 ===\n";
    station.setMeasurements(35.2f, 65.0f, 1005.0f);
    std::cout << "---\n";
    station.setMeasurements(41.5f, 72.3f, 978.0f);
    std::cout << "---\n";

    station.detach(&warning);
    station.setMeasurements(28.0f, 55.0f, 1012.0f);

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from typing import List, Callable, Any
from dataclasses import dataclass
from enum import Enum


# ========== 使用回调函数实现观察者模式（Pythonic 方式） ==========
class EventEmitter:
    """Python 风格的事件发射器"""

    def __init__(self) -> None:
        self._listeners: dict[str, list[Callable]] = {}

    def on(self, event: str, callback: Callable) -> None:
        self._listeners.setdefault(event, []).append(callback)

    def off(self, event: str, callback: Callable) -> None:
        if event in self._listeners:
            self._listeners[event].remove(callback)

    def emit(self, event: str, *args, **kwargs) -> None:
        for callback in self._listeners.get(event, []):
            callback(*args, **kwargs)


# ========== 经典面向对象方式 ==========
class IObserver(ABC):
    @abstractmethod
    def update(self, event: str, data: Any) -> None:
        pass


class Subject:
    def __init__(self) -> None:
        self._observers: List[IObserver] = []
        self._state: str = ""

    def attach(self, observer: IObserver) -> None:
        self._observers.append(observer)

    def detach(self, observer: IObserver) -> None:
        self._observers.remove(observer)

    def notify(self, event: str, data: Any = None) -> None:
        for observer in self._observers:
            observer.update(event, data)

    @property
    def state(self) -> str:
        return self._state

    @state.setter
    def state(self, value: str) -> None:
        self._state = value
        self.notify("state_changed", value)


# ========== 实际应用：股票行情系统 ==========
class Stock:
    def __init__(self, symbol: str, price: float) -> None:
        self.symbol = symbol
        self._price = price
        self._observers: List["StockObserver"] = []

    @property
    def price(self) -> float:
        return self._price

    @price.setter
    def price(self, value: float) -> None:
        old_price = self._price
        self._price = value
        change = value - old_price
        pct = (change / old_price * 100) if old_price else 0
        print(f"📈 {self.symbol}: ¥{old_price:.2f} → ¥{value:.2f} ({pct:+.2f}%)")
        self._notify_observers(value, change, pct)

    def attach(self, observer: "StockObserver") -> None:
        self._observers.append(observer)

    def detach(self, observer: "StockObserver") -> None:
        self._observers.remove(observer)

    def _notify_observers(self, price: float, change: float, pct: float) -> None:
        for obs in self._observers:
            obs.on_price_change(self.symbol, price, change, pct)


class StockObserver(ABC):
    @abstractmethod
    def on_price_change(self, symbol: str, price: float,
                        change: float, pct: float) -> None:
        pass


class PriceAlert(StockObserver):
    def __init__(self, symbol: str, high: float, low: float) -> None:
        self._symbol = symbol
        self._high = high
        self._low = low

    def on_price_change(self, symbol: str, price: float,
                        change: float, pct: float) -> None:
        if symbol != self._symbol:
            return
        if price >= self._high:
            print(f"  🔔 达到高价警戒线 ¥{self._high}: 建议卖出！")
        elif price <= self._low:
            print(f"  🔔 达到低价买入线 ¥{self._low}: 建议买入！")


class TrendAnalyzer(StockObserver):
    def __init__(self) -> None:
        self._history: dict[str, list[float]] = {}

    def on_price_change(self, symbol: str, price: float,
                        change: float, pct: float) -> None:
        self._history.setdefault(symbol, []).append(price)
        history = self._history[symbol]
        if len(history) >= 3:
            last_3 = history[-3:]
            if last_3 == sorted(last_3):
                print(f"  📊 {symbol}: 连续 3 次上涨趋势")
            elif last_3 == sorted(last_3, reverse=True):
                print(f"  📊 {symbol}: 连续 3 次下跌趋势")


if __name__ == "__main__":
    print("=== 回调式事件发射器 ===\n")
    bus = EventEmitter()

    def on_login(user: str) -> None:
        print(f"  🎉 欢迎 {user} 登录！")

    def on_login_log(user: str) -> None:
        print(f"  📝 登录日志: {user} at 2025-09-24")

    bus.on("login", on_login)
    bus.on("login", on_login_log)
    bus.emit("login", "秋鹏")

    print("\n=== 经典观察者：股票系统 ===\n")
    apple = Stock("AAPL", 150.0)
    alert = PriceAlert("AAPL", high=180.0, low=130.0)
    trend = TrendAnalyzer()
    apple.attach(alert)
    apple.attach(trend)

    # 模拟价格变动
    apple.price = 155.0
    apple.price = 160.0
    apple.price = 165.0
    apple.price = 128.0  # 触发低价警报

适用场景

  • 一个对象的改变需要同时改变其他对象，且不知道有多少对象需要改变
  • 事件处理系统：GUI 事件、消息队列、发布订阅
  • 数据绑定：MVVM 框架中的 ViewModel 通知 View 更新
  • 监控系统：指标变化时通知多个告警通道

优点

  • 开闭原则：新增观察者无需修改主题代码
  • 运行时动态：可以动态添加、移除观察者
  • 松耦合：主题只知道观察者接口，不关心具体实现
  • 广播通信：主题一次通知可以触发多个独立行为

缺点

  • 通知顺序不确定：观察者的执行顺序可能影响结果
  • 性能问题：大量观察者时通知可能成为瓶颈
  • 内存泄漏：观察者未正确移除时可能导致引用泄漏
  • 循环依赖：观察者修改主题状态可能触发无限循环通知

与其他模式的关系

  • 观察者 vs 发布订阅：观察者模式中主题直接通知观察者；发布订阅通过事件总线解耦
  • 观察者 vs 中介者：观察者是一对多通知，中介者是多对多协调
  • 观察者 vs 命令模式：观察者被动接收通知，命令模式主动封装请求

实际应用

  • RxJava/RxJS：响应式编程的核心就是观察者模式
  • Vue.js：数据劫持 + 观察者实现响应式更新
  • React：useState/useReducer 的重新渲染机制
  • Java java.util.Observable：JDK 内置的观察者实现（已废弃，推荐 PropertyChangeListener）
  • Python asyncio：事件循环中的回调注册
  • Spring ApplicationEvent：应用内事件发布订阅机制
  • DOM 事件：addEventListener 就是观察者模式
