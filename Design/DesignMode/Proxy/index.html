代理模式

约 1882 字大约 6 分钟

2025-09-24

定义

代理模式为其他对象提供一种代理以控制对这个对象的访问。代理对象在客户端和目标对象之间起到中介作用，可以在请求转发前后添加额外的处理逻辑，如权限校验、延迟加载、日志记录等。

结构

  • Subject（抽象主题）：定义真实主题和代理的公共接口，客户端面向此接口编程。
  • RealSubject（真实主题）：实际执行业务逻辑的对象。
  • Proxy（代理）：持有真实主题的引用，控制对真实主题的访问，在请求前后添加额外处理。
  • Client（客户端）：通过代理间接访问真实主题。

代理模式有多种变体：

  • 远程代理：为远程对象提供本地代表（如 RPC stub）
  • 虚拟代理：延迟创建开销大的对象（如图片懒加载）
  • 保护代理：控制访问权限
  • 缓存代理：缓存请求结果
  • 智能引用代理：在访问对象时执行额外操作（如计数、锁）

UML 类图（文本描述）

Subject <|-- RealSubject
Subject <|-- Proxy
Proxy --> RealSubject（组合/关联：持有真实主题引用）
Client --> Subject（依赖：面向 Subject 接口编程）
  • Proxy 和 RealSubject 都实现 Subject 接口
  • Proxy 内部持有 RealSubject 的引用
  • 客户端通过 Proxy 访问 RealSubject，Proxy 可以控制访问时机和权限

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <memory>
#include <unordered_map>

// ========== 抽象主题 ==========
class IImage {
public:
    virtual ~IImage() = default;
    virtual void display() = 0;
    virtual std::string getFileName() const = 0;
};

// ========== 真实主题 ==========
class RealImage : public IImage {
private:
    std::string fileName_;

    void loadFromDisk() {
        std::cout << "  [RealImage] 正在从磁盘加载: " << fileName_ << std::endl;
        // 模拟耗时的磁盘加载操作
    }

public:
    explicit RealImage(const std::string& fileName)
        : fileName_(fileName) {
        loadFromDisk();
    }

    void display() override {
        std::cout << "  [RealImage] 显示图片: " << fileName_ << std::endl;
    }

    std::string getFileName() const override {
        return fileName_;
    }
};

// ========== 虚拟代理（延迟加载） ==========
class ProxyImage : public IImage {
private:
    std::string fileName_;
    std::unique_ptr<RealImage> realImage_;

    void ensureLoaded() {
        if (!realImage_) {
            realImage_ = std::make_unique<RealImage>(fileName_);
        }
    }

public:
    explicit ProxyImage(const std::string& fileName)
        : fileName_(fileName) {}

    void display() override {
        ensureLoaded();
        realImage_->display();
    }

    std::string getFileName() const override {
        return fileName_;
    }
};

// ========== 保护代理（权限控制） ==========
class IFile {
public:
    virtual ~IFile() = default;
    virtual void read(const std::string& user) = 0;
    virtual void write(const std::string& user, const std::string& content) = 0;
};

class RealFile : public IFile {
private:
    std::string content_;
    std::string name_;
public:
    explicit RealFile(const std::string& name) : name_(name) {}

    void read(const std::string& user) override {
        std::cout << "  [RealFile] " << user << " 读取 " << name_
                  << "，内容: " << content_ << std::endl;
    }
    void write(const std::string& user, const std::string& content) override {
        content_ = content;
        std::cout << "  [RealFile] " << user << " 写入 " << name_ << std::endl;
    }
};

class FileProxy : public IFile {
private:
    std::unique_ptr<RealFile> realFile_;
    bool admin_;

public:
    explicit FileProxy(const std::string& name, bool isAdmin = false)
        : realFile_(std::make_unique<RealFile>(name)), admin_(isAdmin) {}

    void read(const std::string& user) override {
        std::cout << "  [FileProxy] 权限检查: " << user << " 读取" << std::endl;
        realFile_->read(user);
    }
    void write(const std::string& user, const std::string& content) override {
        if (!admin_) {
            std::cout << "  [FileProxy] ✗ 拒绝写入: " << user << " 没有写入权限" << std::endl;
            return;
        }
        std::cout << "  [FileProxy] 权限检查: " << user << " 写入 (✓)" << std::endl;
        realFile_->write(user, content);
    }
};

// ========== 缓存代理 ==========
class ICalculator {
public:
    virtual ~ICalculator() = default;
    virtual int compute(int n) = 0;
};

class HeavyCalculator : public ICalculator {
public:
    int compute(int n) override {
        std::cout << "  [HeavyCalculator] 执行复杂计算: fib(" << n << ")" << std::endl;
        // 简化：模拟耗时计算
        int a = 0, b = 1;
        for (int i = 0; i < n; ++i) { int t = a + b; a = b; b = t; }
        return a;
    }
};

class CachingProxy : public ICalculator {
private:
    std::unique_ptr<HeavyCalculator> realCalc_;
    std::unordered_map<int, int> cache_;

public:
    int compute(int n) override {
        auto it = cache_.find(n);
        if (it != cache_.end()) {
            std::cout << "  [CachingProxy] 缓存命中: fib(" << n << ") = " << it->second << std::endl;
            return it->second;
        }
        if (!realCalc_) realCalc_ = std::make_unique<HeavyCalculator>();
        int result = realCalc_->compute(n);
        cache_[n] = result;
        return result;
    }
};

int main() {
    // 虚拟代理：延迟加载
    std::cout << "=== 虚拟代理（图片懒加载） ===" << std::endl;
    ProxyImage img1("photo1.jpg");
    ProxyImage img2("photo2.jpg");
    std::cout << "代理创建完成，尚未加载图片\n";
    img1.display();  // 此时才真正加载
    std::cout << "---\n";
    img1.display();  // 已加载，不再重复加载
    std::cout << "---\n";
    img2.display();

    // 保护代理
    std::cout << "\n=== 保护代理（权限控制） ===" << std::endl;
    FileProxy config("config.txt", false);
    config.read("普通用户");
    config.write("普通用户", "malicious data");

    FileProxy adminFile("admin_config.txt", true);
    adminFile.write("管理员", "new settings");

    // 缓存代理
    std::cout << "\n=== 缓存代理 ===" << std::endl;
    CachingProxy cacheCalc;
    std::cout << "第一次计算: " << cacheCalc.compute(10) << std::endl;
    std::cout << "第二次计算（缓存命中）: " << cacheCalc.compute(10) << std::endl;

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from functools import wraps
import time


# ========== 抽象主题 ==========
class Subject(ABC):
    @abstractmethod
    def request(self) -> str:
        pass


# ========== 真实主题 ==========
class RealSubject(Subject):
    def request(self) -> str:
        return "RealSubject: 处理请求"


# ========== 代理 ==========
class Proxy(Subject):
    def __init__(self, real_subject: RealSubject) -> None:
        self._real_subject = real_subject

    def _check_access(self) -> bool:
        print("[Proxy] 检查访问权限... ✓")
        return True

    def _log(self) -> None:
        print("[Proxy] 记录访问日志")

    def request(self) -> str:
        if self._check_access():
            self._log()
            result = self._real_subject.request()
            print("[Proxy] 请求处理完成")
            return result
        return "[Proxy] 拒绝访问"


# ========== Python 特有：用装饰器实现代理 ==========
class BankAccount:
    def __init__(self, balance: float) -> None:
        self._balance = balance

    def _verify_owner(self, owner: str) -> bool:
        """验证账户所有者"""
        return owner in ("alice", "admin")

    def withdraw(self, owner: str, amount: float) -> bool:
        if not self._verify_owner(owner):
            print(f"  ✗ 验证失败: {owner} 不是账户所有者")
            return False
        if amount > self._balance:
            print(f"  ✗ 余额不足: {amount} > {self._balance}")
            return False
        self._balance -= amount
        print(f"  ✓ 取款成功: {amount}，余额: {self._balance}")
        return True

    def deposit(self, amount: float) -> None:
        self._balance += amount
        print(f"  ✓ 存款成功: {amount}，余额: {self._balance}")


def transactional(method):
    """事务装饰器：模拟事务代理"""
    @wraps(method)
    def wrapper(self, *args, **kwargs):
        print(f"  [事务代理] 开始事务")
        try:
            result = method(self, *args, **kwargs)
            print(f"  [事务代理] 提交事务")
            return result
        except Exception as e:
            print(f"  [事务代理] 回滚事务: {e}")
            raise
    return wrapper


BankAccount.withdraw = transactional(BankAccount.withdraw)


# ========== 远程代理示例 ==========
class RemoteService:
    """模拟远程服务（实际应通过网络调用）"""
    def query(self, key: str) -> str:
        time.sleep(0.1)  # 模拟网络延迟
        return f"远程数据: {key}"


class RemoteServiceProxy:
    """远程代理：隐藏远程调用的复杂性"""
    def __init__(self) -> None:
        self._service = RemoteService()
        self._connected = False

    def _connect(self) -> None:
        if not self._connected:
            print("[远程代理] 建立连接...")
            self._connected = True

    def query(self, key: str) -> str:
        self._connect()
        print(f"[远程代理] 转发请求: query({key})")
        return self._service.query(key)


if __name__ == "__main__":
    # 基础代理
    print("=== 基础代理 ===")
    real = RealSubject()
    proxy = Proxy(real)
    print(proxy.request())

    # 银行事务代理
    print("\n=== 事务代理 ===")
    account = BankAccount(1000.0)
    account.deposit(500)
    account.withdraw("alice", 200)
    account.withdraw("bob", 100)

    # 远程代理
    print("\n=== 远程代理 ===")
    remote_proxy = RemoteServiceProxy()
    print(remote_proxy.query("user:123"))
    print(remote_proxy.query("order:456"))

适用场景

  • 需要延迟创建开销大的对象（虚拟代理）
  • 需要控制对对象的访问权限（保护代理）
  • 需要为远程对象提供本地代表（远程代理）
  • 需要缓存计算结果或网络请求（缓存代理）
  • 需要记录方法调用日志或性能监控（日志/监控代理）

优点

  • 开闭原则：可以在不修改目标类的情况下增强功能
  • 职责分离：控制逻辑与业务逻辑解耦
  • 延迟加载：虚拟代理可以推迟重量级对象的创建
  • 安全性：保护代理可以细粒度控制访问权限

缺点

  • 性能开销：增加了一层间接调用
  • 复杂度提升：系统中类数量增加
  • 调试困难：请求链路经过代理层，堆栈追踪变长

与其他模式的关系

  • 代理 vs 适配器：代理保持相同接口，适配器改变接口
  • 代理 vs 装饰器：代理控制访问（可能不转发请求），装饰器总是转发并增强
  • 代理 vs 外观模式：代理一对一包装，外观多对一简化

实际应用

  • Spring AOP：通过动态代理实现面向切面编程（事务、日志、安全）
  • Java RMI / gRPC：远程代理隐藏远程调用的网络细节
  • Python unittest.mock：Mock 对象本质上是测试代理
  • C# Lazy<T>：延迟加载代理，首次访问时才创建真实对象
  • 前端图片懒加载：<img loading="lazy"> 是虚拟代理的 HTML 原生实现
  • Hibernate 延迟加载：通过代理实现关联对象的按需加载
