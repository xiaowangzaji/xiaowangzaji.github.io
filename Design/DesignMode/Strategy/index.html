策略模式

约 1676 字大约 6 分钟

2025-09-24

定义

策略模式定义了一系列算法，并将每个算法封装起来，使它们可以互相替换。策略模式让算法的变化独立于使用算法的客户端。核心思想是"用组合代替条件分支"。

结构

  • Strategy（策略接口）：定义所有支持的算法的公共接口。
  • ConcreteStrategy（具体策略）：实现 Strategy 接口，封装具体的算法实现。
  • Context（上下文）：持有一个 Strategy 引用，通过它将请求委托给具体的策略对象。可以在运行时动态切换策略。

策略模式最显著的特征是：将 if-else/switch 分支中的每个分支提取为独立的策略类，通过运行时选择策略来决定执行哪个算法。

UML 类图（文本描述）

Strategy <|-- ConcreteStrategyA
Strategy <|-- ConcreteStrategyB
Strategy <|-- ConcreteStrategyC
Context --> Strategy（组合/关联：持有策略引用）
Client --> Context（通过上下文使用策略）
  • Context 在运行时可以更换 Strategy 实例
  • 客户端通过 Context 间接调用策略，Context 委托给当前策略执行

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <memory>
#include <functional>

// ========== 策略接口：排序 ==========
class SortStrategy {
public:
    virtual ~SortStrategy() = default;
    virtual void sort(std::vector<int>& data) = 0;
    virtual std::string name() const = 0;
};

class QuickSort : public SortStrategy {
public:
    void sort(std::vector<int>& data) override {
        std::cout << "  [快速排序] 对 " << data.size() << " 个元素排序\n";
        std::sort(data.begin(), data.end());  // 使用 std::sort 模拟
    }
    std::string name() const override { return "QuickSort"; }
};

class MergeSort : public SortStrategy {
public:
    void sort(std::vector<int>& data) override {
        std::cout << "  [归并排序] 对 " << data.size() << " 个元素排序\n";
        std::stable_sort(data.begin(), data.end());
    }
    std::string name() const override { return "MergeSort"; }
};

class BubbleSort : public SortStrategy {
public:
    void sort(std::vector<int>& data) override {
        std::cout << "  [冒泡排序] 对 " << data.size() << " 个元素排序\n";
        for (size_t i = 0; i < data.size(); ++i)
            for (size_t j = 0; j < data.size() - 1 - i; ++j)
                if (data[j] > data[j + 1])
                    std::swap(data[j], data[j + 1]);
    }
    std::string name() const override { return "BubbleSort"; }
};

// ========== 上下文 ==========
class Sorter {
private:
    std::unique_ptr<SortStrategy> strategy_;
public:
    void setStrategy(std::unique_ptr<SortStrategy> strategy) {
        strategy_ = std::move(strategy);
    }
    void sort(std::vector<int>& data) {
        if (!strategy_) {
            std::cerr << "未设置排序策略！\n";
            return;
        }
        std::cout << "使用策略: " << strategy_->name() << "\n";
        strategy_->sort(data);
    }
};

// ========== 实际应用：折扣计算 ==========
class DiscountStrategy {
public:
    virtual ~DiscountStrategy() = default;
    virtual double calculate(double originalPrice) = 0;
    virtual std::string description() const = 0;
};

class NoDiscount : public DiscountStrategy {
public:
    double calculate(double price) override { return price; }
    std::string description() const override { return "无折扣"; }
};

class PercentageDiscount : public DiscountStrategy {
private:
    double percent_;
public:
    explicit PercentageDiscount(double percent) : percent_(percent) {}
    double calculate(double price) override { return price * (1 - percent_ / 100.0); }
    std::string description() const override {
        return percent_ == 20 ? "8折优惠" : percent_ == 10 ? "9折优惠" : "折扣";
    }
};

class FixedDiscount : public DiscountStrategy {
private:
    double amount_;
public:
    explicit FixedDiscount(double amount) : amount_(amount) {}
    double calculate(double price) override { return std::max(0.0, price - amount_); }
    std::string description() const override { return "满减"; }
};

class VipDiscount : public DiscountStrategy {
public:
    double calculate(double price) override {
        double after = price * 0.7;
        if (after > 500) after = 500;  // 封顶
        return after;
    }
    std::string description() const override { return "VIP 7折（封顶500）"; }
};

class ShoppingCart {
private:
    std::unique_ptr<DiscountStrategy> discount_;
    double total_ = 0;
public:
    void addItem(double price) { total_ += price; }
    void setDiscount(std::unique_ptr<DiscountStrategy> d) { discount_ = std::move(d); }
    double checkout() {
        if (!discount_) return total_;
        double final_price = discount_->calculate(total_);
        std::cout << "  原价: ¥" << total_
                  << " | " << discount_->description()
                  << " | 折后: ¥" << final_price << "\n";
        return final_price;
    }
};

int main() {
    // 排序策略
    Sorter sorter;
    std::vector<int> data = {5, 2, 9, 1, 7, 3, 8, 4, 6};

    sorter.setStrategy(std::make_unique<QuickSort>());
    sorter.sort(data);

    data = {5, 2, 9, 1, 7};
    sorter.setStrategy(std::make_unique<BubbleSort>());
    sorter.sort(data);

    std::cout << "---\n";

    // 折扣策略
    std::cout << "=== 折扣策略 ===\n";
    ShoppingCart cart;
    cart.addItem(299);
    cart.addItem(199);
    cart.addItem(500);

    cart.setDiscount(std::make_unique<NoDiscount>());
    cart.checkout();

    cart.setDiscount(std::make_unique<PercentageDiscount>(20));
    cart.checkout();

    cart.setDiscount(std::make_unique<FixedDiscount>(100));
    cart.checkout();

    cart.setDiscount(std::make_unique<VipDiscount>());
    cart.checkout();

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Protocol


# ========== 使用 Protocol（Python 3.8+ 鸭子类型策略） ==========
class CompressionStrategy(Protocol):
    def compress(self, data: bytes) -> bytes: ...
    def decompress(self, data: bytes) -> bytes: ...
    @property
    def name(self) -> str: ...


class ZipCompression:
    @property
    def name(self) -> str:
        return "ZIP"

    def compress(self, data: bytes) -> bytes:
        import zipfile, io
        buf = io.BytesIO()
        with zipfile.ZipFile(buf, 'w', zipfile.ZIP_DEFLATED) as zf:
            zf.writestr("data", data)
        return buf.getvalue()

    def decompress(self, data: bytes) -> bytes:
        import zipfile, io
        buf = io.BytesIO(data)
        with zipfile.ZipFile(buf, 'r') as zf:
            return zf.read("data")


class RarCompression:
    @property
    def name(self) -> str:
        return "RAR"

    def compress(self, data: bytes) -> bytes:
        print(f"  [{self.name}] 压缩 {len(data)} 字节")
        return b"[RAR]" + data

    def decompress(self, data: bytes) -> bytes:
        print(f"  [{self.name}] 解压")
        return data[5:]


class SevenZCompression:
    @property
    def name(self) -> str:
        return "7Z"

    def compress(self, data: bytes) -> bytes:
        print(f"  [{self.name}] 压缩 {len(data)} 字节")
        return b"[7Z]" + data

    def decompress(self, data: bytes) -> bytes:
        print(f"  [{self.name}] 解压")
        return data[4:]


class Compressor:
    def __init__(self, strategy: CompressionStrategy) -> None:
        self._strategy = strategy

    def set_strategy(self, strategy: CompressionStrategy) -> None:
        self._strategy = strategy

    def compress_file(self, filename: str) -> bytes:
        with open(filename, 'rb') as f:
            data = f.read()
        print(f"压缩文件: {filename} (原始大小: {len(data)} 字节)")
        return self._strategy.compress(data)


# ========== 实际应用：路由策略 ==========
class RouteStrategy(ABC):
    @abstractmethod
    def build_route(self, start: str, end: str) -> str:
        pass


class WalkingRoute(RouteStrategy):
    def build_route(self, start: str, end: str) -> str:
        return f"🚶 步行路线: {start} → {end} (约 2.5km, 30 分钟)"


class CyclingRoute(RouteStrategy):
    def build_route(self, start: str, end: str) -> str:
        return f"🚲 骑行路线: {start} → {end} (约 2.5km, 10 分钟)"


class DrivingRoute(RouteStrategy):
    def build_route(self, start: str, end: str) -> str:
        return f"🚗 驾车路线: {start} → {end} (约 3.2km, 8 分钟, 含红绿灯)"


class PublicTransitRoute(RouteStrategy):
    def build_route(self, start: str, end: str) -> str:
        return f"🚌 公交路线: {start} → {end} (换乘 1 次, 约 25 分钟, ¥2)"


class Navigator:
    def __init__(self) -> None:
        self._strategy: RouteStrategy = WalkingRoute()

    def set_strategy(self, strategy: RouteStrategy) -> None:
        self._strategy = strategy

    def navigate(self, start: str, end: str) -> str:
        return self._strategy.build_route(start, end)

    nav = Navigator()

        WalkingRoute(), CyclingRoute(), DrivingRoute(), PublicTransitRoute()
    ]
    for s in strategies:
        nav.set_strategy(s)
        print(nav.navigate("公司", "家"))

适用场景

  • 同一类问题有多种不同的算法，需要在运行时切换
  • 用条件分支（if-else/switch）选择算法，且分支可能继续增加
  • 需要封装算法的实现细节，对客户端隐藏复杂度
  • 需要在运行时动态改变对象的行为

优点

  • 开闭原则：新增策略无需修改上下文代码
  • 避免条件分支：用多态代替 if-else/switch
  • 算法可复用：策略可以独立于上下文被其他模块使用
  • 运行时切换：可以在运行时动态更换策略

缺点

  • 类数量增加：每个策略都是一个独立的类
  • 客户端需了解策略差异：客户端需要知道不同策略的适用场景
  • 过度设计：如果只有一个算法，不需要策略模式

与其他模式的关系

  • 策略 vs 状态：两者结构相似，策略关注算法替换，状态关注状态转换
  • 策略 vs 命令：命令封装请求（做什么），策略封装算法（怎么做）
  • 策略 vs 模板方法：策略用组合，模板方法用继承

实际应用

  • Java Comparator：Collections.sort(list, comparator) 中的比较器就是策略
  • Spring Resource：不同的资源加载策略（ClassPathResource、FileSystemResource）
  • Python sorted(key=...)：key 函数就是排序策略
  • 支付方式：支付宝、微信、银行卡、信用卡都是支付策略
  • 压缩算法：ZIP、GZIP、BZ2、LZMA 等
  • 路由算法：Dijkstra、A*、BFS 等路径查找策略
