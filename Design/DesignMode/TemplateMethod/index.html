模板方法模式

约 1838 字大约 6 分钟

2025-09-24

定义

模板方法模式定义一个操作中的算法骨架，将某些步骤延迟到子类中实现。这样，子类可以在不改变算法结构的前提下，重新定义算法中的某些特定步骤。核心思想是"定义骨架，填充细节"。

结构

  • AbstractClass（抽象类）：定义算法的骨架（模板方法），包含一系列步骤方法。其中一些步骤是抽象的，由子类实现；一些步骤有默认实现，子类可以选择覆盖。
  • ConcreteClass（具体类）：实现抽象类中的抽象步骤，完成特定于子类的逻辑。

模板方法通常包含三种方法：

  1. 抽象方法（Abstract Methods）：子类必须实现
  2. 具体方法（Concrete Methods）：抽象类提供默认实现
  3. 钩子方法（Hook Methods）：可选覆盖的默认方法，用于微调算法

UML 类图（文本描述）

AbstractClass
    |-- templateMethod() { step1(); step2(); step3(); }   // 模板方法（final）
    |-- step1() = 0                                        // 抽象方法
    |-- step2() { /* 默认实现 */ }                          // 具体方法（可覆盖）
    |-- step3() = 0                                        // 抽象方法
    |-- hook() { /* 空实现 */ }                            // 钩子方法

AbstractClass <|-- ConcreteClassA
AbstractClass <|-- ConcreteClassB
  • 模板方法通常是 final 的，防止子类篡改算法骨架
  • 子类只能覆盖具体步骤，不能改变整体流程

代码实现（C++ 示例）

#include <iostream>
#include <string>

// ========== 抽象类：数据分析报告 ==========
class DataAnalysisReport {
protected:
    // 钩子方法：是否需要可视化
    virtual bool needsVisualization() const { return true; }

    // 抽象方法：子类必须实现
    virtual void loadData() = 0;
    virtual void processData() = 0;
    virtual void generateInsights() = 0;

    // 具体方法：可覆盖
    virtual void formatOutput() {
        std::cout << "  [格式化] 使用标准文本格式\n";
    }

public:
    virtual ~DataAnalysisReport() = default;

    // 模板方法：定义分析流程
    void generateReport() {
        std::cout << "\n>>> 开始生成分析报告\n";
        loadData();
        processData();
        generateInsights();
        if (needsVisualization()) {
            generateVisualization();
        }
        formatOutput();
        std::cout << ">>> 报告生成完成\n";
    }

private:
    void generateVisualization() {
        std::cout << "  [可视化] 生成图表\n";
    }
};

// ========== 具体类：销售报告 ==========
class SalesReport : public DataAnalysisReport {
private:
    void loadData() override {
        std::cout << "  [数据加载] 从销售数据库读取数据\n";
    }
    void processData() override {
        std::cout << "  [数据处理] 计算销售额、利润率、增长率\n";
    }
    void generateInsights() override {
        std::cout << "  [洞察] Q3 销售额同比增长 15%，华东区域最佳\n";
    }
    void formatOutput() override {
        std::cout << "  [格式化] 输出为 PDF 格式\n";
    }
};

// ========== 具体类：简单日志分析 ==========
class LogAnalysisReport : public DataAnalysisReport {
private:
    void loadData() override {
        std::cout << "  [数据加载] 从日志文件读取数据\n";
    }
    void processData() override {
        std::cout << "  [数据处理] 统计错误数和请求数\n";
    }
    void generateInsights() override {
        std::cout << "  [洞察] 500 错误占比 2%，高于正常阈值\n";
    }
    bool needsVisualization() const override {
        return false;  // 简单日志报告不需要可视化
    }
};

// ========== 实际应用：游戏 AI 行为 ==========
class GameAI {
protected:
    virtual void collectResources() = 0;
    virtual void buildUnits() = 0;
    virtual void attack() = 0;
    virtual bool isRushStrategy() const { return false; }  // 钩子

public:
    virtual ~GameAI() = default;

    // 模板方法：AI 回合
    void playTurn() {
        if (isRushStrategy()) {
            std::cout << "  ⚡ [策略] rush! 快速进攻\n";
            collectResources();
            buildUnits();
            attack();
        } else {
            std::cout << "  🏗 [策略] 稳扎稳打\n";
            collectResources();
            buildUnits();
            // 发展回合先不进攻
            std::cout << "  🛡 本回合防守\n";
        }
    }
};

class OrcAI : public GameAI {
private:
    void collectResources() override {
        std::cout << "  [兽人] 掠夺资源\n";
    }
    void buildUnits() override {
        std::cout << "  [兽人] 训练战士\n";
    }
    void attack() override {
        std::cout << "  [兽人] 全面进攻！\n";
    }
    bool isRushStrategy() const override { return true; }
};

class ElfAI : public GameAI {
private:
    void collectResources() override {
        std::cout << "  [精灵] 采集资源\n";
    }
    void buildUnits() override {
        std::cout << "  [精灵] 训练弓箭手\n";
    }
    void attack() override {
        std::cout << "  [精灵] 远程射击\n";
    }
};

int main() {
    std::cout << "=== 模板方法：数据分析 ===\n";
    SalesReport salesReport;
    salesReport.generateReport();

    LogAnalysisReport logReport;
    logReport.generateReport();

    std::cout << "\n=== 模板方法：游戏 AI ===\n";
    OrcAI orc;
    ElfAI elf;
    orc.playTurn();
    std::cout << "---\n";
    elf.playTurn();

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod


# ========== 抽象类：数据导出器 ==========
class DataExporter(ABC):
    """模板方法：数据导出流程"""

    def export(self, source: str) -> None:
        """模板方法：不可被子类覆盖"""
        print(f"\n>>> 开始从 {source} 导出数据")
        self.connect_to_source(source)
        data = self.fetch_data()
        transformed = self.transform_data(data)
        self.write_output(transformed)
        self.disconnect()
        print(">>> 导出完成\n")

    def connect_to_source(self, source: str) -> None:
        """钩子：默认实现，子类可覆盖"""
        print(f"  连接到数据源: {source}")

    @abstractmethod
    def fetch_data(self) -> list[dict]:
        """抽象方法：子类必须实现"""
        pass

    @abstractmethod
    def transform_data(self, data: list[dict]) -> str:
        """抽象方法：子类必须实现"""
        pass

    @abstractmethod
    def write_output(self, content: str) -> None:
        """抽象方法：子类必须实现"""
        pass

    def disconnect(self) -> None:
        """钩子：默认实现"""
        print("  断开连接")


# ========== 具体类：CSV 导出 ==========
class CSVExporter(DataExporter):
    def fetch_data(self) -> list[dict]:
        print("  从数据库查询用户数据")
        return [
            {"name": "张三", "age": 28, "city": "北京"},
            {"name": "李四", "age": 32, "city": "上海"},
        ]

    def transform_data(self, data: list[dict]) -> str:
        if not data:
            return ""
        headers = ",".join(data[0].keys())
        rows = [headers]
        for row in data:
            rows.append(",".join(str(v) for v in row.values()))
        return "\n".join(rows)

    def write_output(self, content: str) -> None:
        print("  写入 CSV 文件:")
        for line in content.split("\n"):
            print(f"    {line}")


# ========== 具体类：JSON 导出 ==========
class JSONExporter(DataExporter):
    def connect_to_source(self, source: str) -> None:
        print(f"  通过 API 连接到: {source}")

    def fetch_data(self) -> list[dict]:
        print("  通过 REST API 获取数据")
        return [
            {"id": 1, "product": "笔记本", "price": 5999},
            {"id": 2, "product": "手机", "price": 3999},
        ]

    def transform_data(self, data: list[dict]) -> str:
        import json
        return json.dumps(data, ensure_ascii=False, indent=2)

    def write_output(self, content: str) -> None:
        print("  写入 JSON 文件:")
        print(f"    {content}")


# ========== 实际应用：Web 请求处理 ==========
class HttpRequestHandler(ABC):
    """模板方法：HTTP 请求处理流程"""

    def handle(self, request: dict) -> dict:
        """模板方法"""
        if not self.authorize(request):
            return {"status": 403, "message": "未授权"}

        if not self.validate(request):
            return {"status": 400, "message": "参数验证失败"}

        data = self.process(request)
        return self.format_response(data)

    def authorize(self, request: dict) -> bool:
        """钩子：默认通过"""
        return True

    def validate(self, request: dict) -> bool:
        """钩子：默认通过"""
        return True

    @abstractmethod
    def process(self, request: dict) -> dict: ...

    def format_response(self, data: dict) -> dict:
        """钩子：默认实现"""
        return {"status": 200, "data": data}


class UserApiHandler(HttpRequestHandler):
    def validate(self, request: dict) -> bool:
        return "username" in request and len(request["username"]) >= 3

    def process(self, request: dict) -> dict:
        return {"user_id": 1001, "username": request["username"], "role": "admin"}


if __name__ == "__main__":
    print("=== 模板方法：数据导出 ===")
    csv_exporter = CSVExporter()
    csv_exporter.export("MySQL:localhost/users")

    json_exporter = JSONExporter()
    json_exporter.export("https://api.example.com/products")

    print("\n=== 模板方法：HTTP 请求 ===")
    handler = UserApiHandler()
    print(handler.handle({"username": "qiupeng"}))
    print(handler.handle({"username": "ab"}))  # 验证失败

适用场景

  • 多个子类有相同的算法骨架，只是某些步骤不同
  • 需要控制子类的扩展点（通过钩子方法）
  • 避免代码重复：将公共逻辑上提到父类
  • 框架设计：Spring 的 JdbcTemplate、Django 的 CBV

优点

  • 代码复用：公共逻辑集中在父类，避免重复
  • 反向控制（好莱坞原则）：父类调用子类，而非子类调用父类
  • 扩展点可控：通过抽象方法和钩子精确控制子类能做什么
  • 流程规范：模板方法保证了算法骨架的一致性

缺点

  • 继承耦合：子类与父类紧密绑定
  • 调试困难：执行流程在父子类之间跳转
  • 灵活性受限：子类只能修改特定步骤，不能改变整体流程

与其他模式的关系

  • 模板方法 vs 策略：模板方法用继承，策略用组合
  • 模板方法 vs 工厂方法：工厂方法是模板方法的特例（只创建一个对象）
  • 模板方法 vs 建造者：模板方法定义固定流程，建造者流程由 Director 控制

实际应用

  • Java AbstractList：提供了 Iterator、equals、hashCode 等的模板实现
  • Spring JdbcTemplate：定义数据库操作流程，子类只需实现 RowMapper
  • Django Class-Based Views：get()/post() 是钩子，dispatch() 是模板方法
  • Python abc 模块：抽象基类本身就是模板方法的支撑
  • JUnit TestCase：setUp() → testXxx() → tearDown() 是模板流程
  • HTTP 框架中间件：请求处理管道本质上是模板方法链
