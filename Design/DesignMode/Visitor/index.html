访问者模式

约 1601 字大约 5 分钟

2025-09-24

定义

访问者模式表示一个作用于某对象结构中的各元素的操作。它使你可以在不改变各元素类的前提下定义作用于这些元素的新操作。核心思想是"将数据结构与数据操作分离"。

结构

  • Visitor（访问者接口）：为每种具体元素类声明一个访问方法 visit(ConcreteElement)。方法名通常相同但参数类型不同（方法重载）。
  • ConcreteVisitor（具体访问者）：实现访问者接口，定义对每种元素的具体操作。
  • Element（元素接口）：声明一个 accept(Visitor) 方法，接收访问者对象。
  • ConcreteElement（具体元素）：实现 accept 方法，调用访问者的对应 visit 方法（visitor.visit(this)）。
  • ObjectStructure（对象结构）：元素的集合，提供遍历接口。

访问者模式使用了双重分发（Double Dispatch）：第一次分发是客户端调用 element.accept(visitor)，第二次分发是元素调用 visitor.visit(this)。

UML 类图（文本描述）

Visitor <|-- ConcreteVisitorA
Visitor <|-- ConcreteVisitorB
    |-- visit(ElementA)
    |-- visit(ElementB)

Element <|-- ConcreteElementA
Element <|-- ConcreteElementB
    |-- accept(Visitor) { visitor.visit(this) }

ObjectStructure --> Element（聚合：持有元素集合）

双重分发:
  1. client → element.accept(visitor)    [第一次：选择具体元素类型]
  2. element → visitor.visit(this)       [第二次：选择具体访问者类型]

代码实现（C++ 示例）

#include <iostream>
#include <string>
#include <vector>
#include <memory>
#include <numeric>

// ========== 前向声明 ==========
class ConcreteElementA;
class ConcreteElementB;

// ========== 访问者接口 ==========
class Visitor {
public:
    virtual ~Visitor() = default;
    virtual void visitA(ConcreteElementA* element) = 0;
    virtual void visitB(ConcreteElementB* element) = 0;
};

// ========== 元素接口 ==========
class Element {
public:
    virtual ~Element() = default;
    virtual void accept(Visitor* visitor) = 0;
};

// ========== 具体元素 A：圆形 ==========
class ConcreteElementA : public Element {
private:
    double radius_;
public:
    explicit ConcreteElementA(double radius) : radius_(radius) {}
    double getRadius() const { return radius_; }

    void accept(Visitor* visitor) override {
        visitor->visitA(this);  // 双重分发
    }
};

// ========== 具体元素 B：矩形 ==========
class ConcreteElementB : public Element {
private:
    double width_, height_;
public:
    ConcreteElementB(double w, double h) : width_(w), height_(h) {}
    double getWidth() const { return width_; }
    double getHeight() const { return height_; }

    void accept(Visitor* visitor) override {
        visitor->visitB(this);  // 双重分发
    }
};

// ========== 具体访问者：面积计算 ==========
class AreaCalculator : public Visitor {
public:
    double totalArea = 0;

    void visitA(ConcreteElementA* element) override {
        double area = 3.14159 * element->getRadius() * element->getRadius();
        std::cout << "  圆形面积: π × " << element->getRadius() << "² = "
                  << area << "\n";
        totalArea += area;
    }
    void visitB(ConcreteElementB* element) override {
        double area = element->getWidth() * element->getHeight();
        std::cout << "  矩形面积: " << element->getWidth() << " × "
                  << element->getHeight() << " = " << area << "\n";
        totalArea += area;
    }
};

// ========== 具体访问者：XML 导出 ==========
class XMLExporter : public Visitor {
private:
    std::string xml_;
public:
    std::string getResult() const { return xml_; }

    void visitA(ConcreteElementA* element) override {
        xml_ += "  <circle radius=\"" +
                std::to_string(element->getRadius()) + "\"/>\n";
    }
    void visitB(ConcreteElementB* element) override {
        xml_ += "  <rect width=\"" +
                std::to_string(element->getWidth()) +
                "\" height=\"" +
                std::to_string(element->getHeight()) + "\"/>\n";
    }
};

// ========== 对象结构 ==========
class ShapeCollection {
private:
    std::vector<std::shared_ptr<Element>> elements_;
public:
    void add(std::shared_ptr<Element> elem) {
        elements_.push_back(elem);
    }
    void accept(Visitor* visitor) {
        for (auto& elem : elements_) {
            elem->accept(visitor);
        }
    }
};

int main() {
    ShapeCollection shapes;
    shapes.add(std::make_shared<ConcreteElementA>(5.0));
    shapes.add(std::make_shared<ConcreteElementB>(10.0, 20.0));
    shapes.add(std::make_shared<ConcreteElementA>(3.0));
    shapes.add(std::make_shared<ConcreteElementB>(8.0, 12.0));

    std::cout << "=== 访问者模式：面积计算 ===\n";
    AreaCalculator areaCalc;
    shapes.accept(&areaCalc);
    std::cout << "总面积: " << areaCalc.totalArea << "\n\n";

    std::cout << "=== 访问者模式：XML 导出 ===\n";
    XMLExporter exporter;
    shapes.accept(&exporter);
    std::cout << "<shapes>\n" << exporter.getResult() << "</shapes>\n";

    return 0;
}

代码实现（Python 示例）

from abc import ABC, abstractmethod
from typing import List, Any


# ========== 前向引用：访问者 ==========
class Visitor(ABC):
    @abstractmethod
    def visit_file(self, file_elem: "File") -> Any: ...

    @abstractmethod
    def visit_directory(self, dir_elem: "Directory") -> Any: ...

    @abstractmethod
    def visit_link(self, link_elem: "Link") -> Any: ...


# ========== 元素接口 ==========
class FileSystemEntry(ABC):
    @abstractmethod
    def accept(self, visitor: Visitor) -> Any: ...


# ========== 具体元素：文件 ==========
class File(FileSystemEntry):
    def __init__(self, name: str, size: int) -> None:
        self._name = name
        self._size = size

    @property
    def name(self) -> str:
        return self._name

    @property
    def size(self) -> int:
        return self._size

    def accept(self, visitor: Visitor) -> Any:
        return visitor.visit_file(self)


# ========== 具体元素：目录 ==========
class Directory(FileSystemEntry):
    def __init__(self, name: str) -> None:
        self._name = name
        self._children: List[FileSystemEntry] = []

    @property
    def name(self) -> str:
        return self._name

    def add(self, entry: FileSystemEntry) -> None:
        self._children.append(entry)

    def accept(self, visitor: Visitor) -> Any:
        return visitor.visit_directory(self)

    def iter_children(self):
        return iter(self._children)


# ========== 具体元素：符号链接 ==========
class Link(FileSystemEntry):
    def __init__(self, name: str, target: str) -> None:
        self._name = name
        self._target = target

    @property
    def name(self) -> str:
        return self._name

    @property
    def target(self) -> str:
        return self._target

    def accept(self, visitor: Visitor) -> Any:
        return visitor.visit_link(self)


# ========== 具体访问者：总大小计算 ==========
class SizeCalculatorVisitor(Visitor):
    def __init__(self) -> None:
        self._total_size = 0

    def visit_file(self, file_elem: File) -> int:
        self._total_size += file_elem.size
        return file_elem.size

    def visit_directory(self, dir_elem: Directory) -> int:
        dir_size = 0
        for child in dir_elem.iter_children():
            dir_size += child.accept(self)
        self._total_size += dir_size
        return dir_size

    def visit_link(self, link_elem: Link) -> int:
        return 0  # 链接不占空间

    @property
    def total_size(self) -> int:
        return self._total_size


# ========== 具体访问者：树形显示 ==========
class TreeDisplayVisitor(Visitor):
    def __init__(self) -> None:
        self._output = ""
        self._indent = 0

    def _prefix(self) -> str:
        return "  " * self._indent

    def visit_file(self, file_elem: File) -> None:
        self._output += f"{self._prefix()}📄 {file_elem.name} ({file_elem.size}B)\n"

    def visit_directory(self, dir_elem: Directory) -> None:
        self._output += f"{self._prefix()}📁 {dir_elem.name}/\n"
        self._indent += 1
        for child in dir_elem.iter_children():
            child.accept(self)
        self._indent -= 1

    def visit_link(self, link_elem: Link) -> None:
        self._output += f"{self._prefix()}🔗 {link_elem.name} -> {link_elem.target}\n"

    @property
    def output(self) -> str:
        return self._output


if __name__ == "__main__":
    print("=== 访问者模式：文件系统 ===\n")

    # 构建文件系统树
    root = Directory("home")
    docs = Directory("docs")
    src = Directory("src")

    docs.add(File("README.md", 2048))
    docs.add(File("design.md", 5120))
    src.add(File("main.py", 8192))
    src.add(File("utils.py", 4096))
    root.add(docs)
    root.add(src)
    root.add(File(".gitignore", 128))
    root.add(Link("shortcut", "docs/README.md"))

    # 访问者 1：计算总大小
    size_calc = SizeCalculatorVisitor()
    root.accept(size_calc)
    print(f"文件系统总大小: {size_calc.total_size} 字节\n")

    # 访问者 2：树形显示
    tree_view = TreeDisplayVisitor()
    root.accept(tree_view)
    print(tree_view.output)

    print("--- 访问者模式优势 ---")
    print("新增操作（如搜索、导出、权限检查）只需新增访问者类")
    print("无需修改 File、Directory、Link 等元素类")

适用场景

  • 对象结构稳定（元素类很少变化），但需要频繁新增操作
  • 需要对对象结构中的元素执行多种不同的操作
  • 元素类之间有复杂的交互，需要集中处理
  • 编译器 AST 遍历、语法分析、代码生成

优点

  • 开闭原则（对操作）：新增操作只需添加新的访问者类
  • 集中相关操作：所有与某个操作相关的逻辑集中在一个访问者中
  • 累积状态：访问者可以在遍历过程中累积状态（如总和、计数）
  • 打破封装：可以访问元素对象的内部状态（在需要时）

缺点

  • 违反开闭原则（对元素）：新增元素类型需要修改所有访问者
  • 破坏封装：访问者需要访问元素的内部数据
  • 双重分发开销：多一层方法调用
  • 适用场景受限：只有当元素类稳定时才适合

与其他模式的关系

  • 访问者 vs 迭代器：迭代器遍历结构，访问者在遍历时执行操作
  • 访问者 vs 组合：访问者常与组合模式配合遍历树形结构
  • 访问者 vs 命令：命令封装请求，访问者封装操作

实际应用

  • 编译器：AST 遍历（语法检查、优化、代码生成）
  • Java NIO.2：FileVisitor 接口遍历文件系统
  • ANTLR：生成的语法树访问者
  • Jackson JSON：TreeModel 的遍历
  • Clang/LLVM：AST Visitor 模式
  • ESLint/Prettier：AST 遍历检查/格式化代码
